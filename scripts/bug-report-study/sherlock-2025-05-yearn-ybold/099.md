# H-03: LiquityV2SPStrategy – Division-by-WAD Bug Underprices Collateral Auctions

### Summary

Collateral auction pricing is under-scaled by an extra `/ WAD`, converting a correct WAD-scaled price into an unscaled tiny value. Bots can bid nearly zero and drain all collateral.

### Root Cause

```solidity
// erroneous: three divisions by WAD
AUCTION.setStartingPrice(
    _available * _price / WAD
    * _bufferPercentage / WAD
    / WAD
);
```
Only **two** `/ WAD` operations are needed; the third shrinks the price by 1e18.

### Internal Pre-conditions

- Keeper calls `tend()`.

### External Pre-conditions

- Strategy has collateral gains.  
- Oracle returns WAD-scaled price.  

### Attack Path

1. `tend()` claims `C` collateral.  
2. Calculates `ideal = (C * price / 1e18) * buffer / 1e18`.  
3. Bug applies a final `/ 1e18`, yielding `idealNumeric = ideal / 1e18`.  
4. Auction sees near-zero price and is instantly won.

### Impact

- **Total Loss:** Collateral sold at near-zero price.  
- **Automated Drain:** Bots can repeat exploit each cycle.

### PoC

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.23;

import "./Strategy.sol";           // LiquityV2SPStrategy
import "./interfaces/IPriceFeed.sol";
import "./interfaces/IStabilityPool.sol";
import "./interfaces/IAuction.sol";
import "./interfaces/IAuctionFactory.sol";
import "./interfaces/IAddressesRegistry.sol";
import "@openzeppelin/contracts/token/ERC20/ERC20.sol";

uint256 constant WAD = 1e18;

/// @notice Mocks a simple price feed returning a fixed price
contract MockPriceFeed is IPriceFeed {
    uint256 public price = 2000 * WAD;
    bool public down;

    function fetchPrice() external view override returns (uint256, bool) {
        return (price, down);
    }
    function priceSource() external pure override returns (PriceSource) {
        return PriceSource.primary;
    }
}

/// @notice Mocks StabilityPool to credit collateral gains
contract MockStabilityPool is IStabilityPool {
    address public override collToken;
    mapping(address => uint256) public gains;

    constructor(address _coll) { collToken = _coll; }
    function collToken() external view override returns (address) { return collToken; }
    function boldToken() external pure override returns (address) { return address(0); }
    function getDepositorCollGain(address _d) external view override returns (uint256) {
        return gains[_d];
    }
    function provideToSP(uint256 _amt, bool) external override {}
    function withdrawFromSP(uint256, bool claim) external override {
        if (claim) ERC20(collToken).transfer(msg.sender, gains[msg.sender]);
    }
    function setGain(address who, uint256 amount) external {
        gains[who] = amount;
    }
}

/// @notice Captures the last auction price set
contract MockAuction is IAuction {
    uint256 public lastPrice;
    address public kicked;

    function enable(address) external override {}
    function setStartingPrice(uint256 p) external override { lastPrice = p; }
    function kick(address token) external override { kicked = token; }
    function settle(address) external override {}
    function isActive(address) external pure override returns (bool) { return false; }
    function available(address) external pure override returns (uint256) { return 0; }
}

/// @notice Factory that always returns our MockAuction
contract MockAuctionFactory is IAuctionFactory {
    MockAuction public last;
    function createNewAuction(address) external override returns (IAuction) {
        last = new MockAuction();
        return last;
    }
}

/// @notice Registry stub linking our mocks
contract MockRegistry is IAddressesRegistry {
    IPriceFeed      public pf;
    IStabilityPool  public sp;
    IAuctionFactory public af;

    constructor(IPriceFeed _pf, IStabilityPool _sp, IAuctionFactory _af) {
        pf = _pf; sp = _sp; af = _af;
    }
    function priceFeed() external view override returns (IPriceFeed) { return pf; }
    function stabilityPool() external view override returns (IStabilityPool) { return sp; }
    function auctionFactory() external view override returns (IAuctionFactory) { return af; }
}

/// @notice Demonstrates the underpricing bug in tend()
contract PoC_H03 {
    LiquityV2SPStrategy public strat;
    ERC20 public collToken;
    MockPriceFeed      pf;
    MockStabilityPool  sp;
    MockAuctionFactory af;
    MockRegistry       reg;

    constructor() {
        collToken = new ERC20("COLL", "COLL");
        pf        = new MockPriceFeed();
        sp        = new MockStabilityPool(address(collToken));
        af        = new MockAuctionFactory();
        reg       = new MockRegistry(pf, sp, af);

        strat = new LiquityV2SPStrategy(address(reg), address(collToken), "TestLQStrat");
        strat.setKeeper(address(this));

        sp.setGain(address(strat), 10 * WAD);
        collToken.mint(address(strat), 10 * WAD);
    }

    function runTend() external {
        strat.tend();
        MockAuction auc = af.last;
        uint256 expectedWAD = (((10 * WAD) * (2000 * WAD)) / WAD * (WAD + WAD/10)) / WAD;
        uint256 bugged    = expectedWAD / WAD;
        require(auc.lastPrice == bugged, "Auction price not under-scaled");
        require(auc.kicked == address(collToken), "Wrong token kicked");
    }
}
```
