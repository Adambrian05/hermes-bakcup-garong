# Governance Function Misuse

## Summary
The strategy incorrectly calls the governance-only `setStartingPrice()` function instead of an auction initiation function, causing all auction attempts to fail and rendering the strategy inoperable.

## Description
`Accountant::_tend` attempts to start auctions by calling `AUCTION.setStartingPrice()`:

```solidity
AUCTION.setStartingPrice(_available * _price / WAD * _bufferPercentage / WAD / WAD);
```

However, `setStartingPrice()` is a governance-only configuration function that:
1. Requires `onlyGovernance` modifier - the strategy contract lacks governance permissions
2. Sets a global starting price parameter for all auctions (not auction-specific)
3. Reverts if ANY auction is active across the entire system

This breaks the strategy's core security guarantee of being able to liquidate collateral through auctions. The function will **always revert** with "Unauthorized" (due to governance check) or "active auction" (due to system-wide auction state), making collateral liquidation impossible.

## Impact
**Impact: High**

This completely breaks the strategy's mechanism because no auction can ever be initiated by the strategy

This is not a DoS or temporary issue, it's a fundamental architectural flaw that makes the contract unusable for its intended purpose.

## Likelihood  
**Likelihood: High**

This will happen on **every single call** to `_tend()` that attempts to create an auction:
- The governance check will fail immediately (100% guaranteed)
- Even if governance were granted, any active auction system-wide would cause failure
- No external conditions or malicious actors required
- The bug is deterministic and will manifest in normal operation

## Proof of Concept
1. Strategy accumulates collateral and calls `_tend()`
2. Collateral balance exceeds `dustThreshold` 
3. Strategy calculates starting price and calls `AUCTION.setStartingPrice(calculatedPrice)`
4. Auction contract checks `onlyGovernance` modifier
5. Strategy contract is not governance → transaction reverts with "Unauthorized"
6. No auction is created, collateral remains in strategy
7. This repeats on every subsequent `_tend()` call forever

- export ETH_RPC_URL=<RPC_URL>

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

import "forge-std/Test.sol";

// Mock interfaces
interface IAuction {
    function setStartingPrice(uint256 _startingPrice) external;
    function isActive(address token) external view returns (bool);
    function available(address token) external view returns (uint256);
    function settle(address token) external;
    function kick(address token) external;
}

interface IERC20 {
    function balanceOf(address account) external view returns (uint256);
    function safeTransfer(address to, uint256 amount) external;
}

interface IPriceFeed {
    enum PriceSource { primary, secondary }
    function fetchPrice() external view returns (uint256 price, bool isOracleDown);
    function priceSource() external view returns (PriceSource);
}

// Mock Strategy contract with the vulnerable _tend function
contract VulnerableStrategy {
    IAuction public AUCTION;
    IERC20 public COLL;
    IPriceFeed public COLL_PRICE_ORACLE;
    
    uint256 public dustThreshold = 1000;
    uint256 public bufferPercentage = 95e16; // 95%
    uint256 public constant WAD = 1e18;
    uint256 public constant ORACLE_DOWN_BUFFER_PCT_MULTIPLIER = 2;
    
    constructor(address _auction, address _coll, address _oracle) {
        AUCTION = IAuction(_auction);
        COLL = IERC20(_coll);
        COLL_PRICE_ORACLE = IPriceFeed(_oracle);
    }
    
    function isCollateralGainToClaim() internal pure returns (bool) {
        return false; // Simplified for test
    }
    
    function claim() internal {
        // Simplified for test
    }
    
    // The vulnerable function
    function _tend(uint256 /*_totalIdle*/) internal {
        if (isCollateralGainToClaim()) claim();
        if (AUCTION.isActive(address(COLL)) && AUCTION.available(address(COLL)) == 0) {
            AUCTION.settle(address(COLL));
        }

        uint256 _toAuction = COLL.balanceOf(address(this));
        uint256 _available = COLL.balanceOf(address(AUCTION)) + _toAuction;
        
        if (_available > dustThreshold) {
            (uint256 _price, bool _isOracleDown) = COLL_PRICE_ORACLE.fetchPrice();
            uint256 _bufferPercentage = bufferPercentage;
            
            if (_isOracleDown || COLL_PRICE_ORACLE.priceSource() != IPriceFeed.PriceSource.primary) {
                _bufferPercentage *= ORACLE_DOWN_BUFFER_PCT_MULTIPLIER;
            }

            // THE VULNERABLE LINE - calls governance function instead of auction initiation
            AUCTION.setStartingPrice(_available * _price / WAD * _bufferPercentage / WAD / WAD);
            
            COLL.safeTransfer(address(AUCTION), _toAuction);
            AUCTION.kick(address(COLL));
        }
    }
    
    // Public wrapper for testing
    function tend() external {
        _tend(0);
    }
}

// Mock Auction contract that implements governance restrictions
contract MockAuction {
    address public governance;
    uint256 public startingPrice;
    
    modifier onlyGovernance() {
        require(msg.sender == governance, "not governance");
        _;
    }
    
    constructor() {
        governance = msg.sender; // Test contract is governance
    }
    
    function setStartingPrice(uint256 _startingPrice) external onlyGovernance {
        require(_startingPrice != 0, "starting price");
        startingPrice = _startingPrice;
    }
    
    function isActive(address) external pure returns (bool) {
        return false;
    }
    
    function available(address) external pure returns (uint256) {
        return 0;
    }
    
    function settle(address) external {}
    function kick(address) external {}
    function balanceOf(address) external pure returns (uint256) {
        return 0;
    }
}

// Mock ERC20 token
contract MockERC20 {
    mapping(address => uint256) public balanceOf;
    
    function setBalance(address user, uint256 amount) external {
        balanceOf[user] = amount;
    }
    
    function safeTransfer(address, uint256) external {}
}

// Mock Price Oracle
contract MockPriceOracle {
    function fetchPrice() external pure returns (uint256, bool) {
        return (2000e18, false); // $2000, oracle up
    }
    
    function priceSource() external pure returns (IPriceFeed.PriceSource) {
        return IPriceFeed.PriceSource.primary;
    }
}

contract GovernanceMisuseTest is Test {
    VulnerableStrategy strategy;
    MockAuction auction;
    MockERC20 collToken;
    MockPriceOracle oracle;
    
    function setUp() public {
        auction = new MockAuction();
        collToken = new MockERC20();
        oracle = new MockPriceOracle();
        strategy = new VulnerableStrategy(address(auction), address(collToken), address(oracle));
        
        // Set up collateral balance above dust threshold
        collToken.setBalance(address(strategy), 5000); // Above dustThreshold of 1000
    }
    
    function testGovernanceFunctionReverts() public {
        // The strategy should revert when trying to call governance-only function
        vm.expectRevert("not governance");
        strategy.tend();
    }
    
    function testProofOfConcept() public {
        // Demonstrate that even with sufficient collateral and proper conditions,
        // the strategy cannot function due to governance restriction
        
        uint256 strategyBalance = collToken.balanceOf(address(strategy));
        assertGt(strategyBalance, 1000, "Strategy should have collateral above dust threshold");
        
        // Auction is not active
        assertFalse(auction.isActive(address(collToken)), "No active auction");
        
        // Oracle is working properly  
        (uint256 price, bool isDown) = oracle.fetchPrice();
        assertGt(price, 0, "Oracle should return valid price");
        assertFalse(isDown, "Oracle should be up");
        
        // Despite all conditions being met for auction creation, the call fails
        vm.expectRevert("not governance");
        strategy.tend();
        
        // Strategy remains broken - cannot liquidate collateral
        assertEq(collToken.balanceOf(address(strategy)), strategyBalance, "Collateral stuck in strategy");
    }
    
    function testAlternativeIfStrategyWereGovernance() public {
        // Even if we made strategy the governance (which would be wrong design),
        // let's show it would work
        
        // Deploy new auction with strategy as governance
        vm.prank(address(strategy));
        MockAuction strategyGovAuction = new MockAuction();
        
        // Deploy new strategy with this auction
        VulnerableStrategy strategyAsGov = new VulnerableStrategy(
            address(strategyGovAuction), 
            address(collToken), 
            address(oracle)
        );
        
        collToken.setBalance(address(strategyAsGov), 5000);
        
        // This would work (but is wrong design)
        strategyAsGov.tend();
        
        // Show that the starting price was set
        assertGt(strategyGovAuction.startingPrice(), 0, "Starting price was set");
    }
}

## Recommendation

The strategy should call the appropriate auction initiation function instead of the governance configuration function:

```diff
- AUCTION.setStartingPrice(_available * _price / WAD * _bufferPercentage / WAD / WAD);
+ // Assuming the correct function is kick() or start() - check auction contract docs
+ AUCTION.kick(address(COLL), _available * _price / WAD * _bufferPercentage / WAD / WAD);
  
  COLL.safeTransfer(address(AUCTION), _toAuction);
- AUCTION.kick(address(COLL));
+ // Remove redundant kick call if price is set in the above call
```

Alternative fix if auction requires separate price setting:
```diff
+ // Only set starting price if contract has governance role and no auctions active
+ if (hasGovernanceRole() && !hasActiveAuctions()) {
+     AUCTION.setStartingPrice(_available * _price / WAD * _bufferPercentage / WAD / WAD);
+ }
+ COLL.safeTransfer(address(AUCTION), _toAuction);
  AUCTION.kick(address(COLL));
```
