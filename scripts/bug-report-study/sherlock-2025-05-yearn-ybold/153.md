# Centralized Health Check Bypass

## Summary
The fee manager in `Accountant::turnOffHealthCheck` can instantly disable health checks for any vault/strategy combination without time delays or transparency, creating a centralized point of failure that can lead to catastrophic vault losses.

## Description
The `turnOffHealthCheck` function allows the fee manager to bypass critical safety mechanisms that prevent strategies from reporting excessive losses. The health check normally enforces:
- Maximum allowable losses via `fee.maxLoss` 
- Maximum allowable gains via `fee.maxGain`

When bypassed, a strategy can report unlimited losses without the transaction reverting. The bypass is implemented through a simple boolean flag `skipHealthCheck[vault][strategy]` that completely disables the require statements in the report function.

A malicious or compromised fee manager could:
1. Call `turnOffHealthCheck(vault, strategy)` for a target vault
2. The strategy can then report 100% loss of `current_debt` 
3. Vault users lose their entire deposits with no warning or protection


## Impact
**High Impact** - This vulnerability can result in complete loss of user funds in affected vaults. The health check is a core safety mechanism designed to prevent catastrophic losses from strategy bugs, market manipulation, or malicious behavior. When bypassed, strategies can drain entire vault balances. Given that vaults typically hold substantial user deposits, the financial impact is severe.

The impact is amplified because:
- Users have no visibility into when protections are disabled
- No automatic safeguards prevent abuse
- Multiple vaults can be affected simultaneously

## Likelihood
**Medium Likelihood** - While this requires the fee manager to be malicious or compromised, single points of failure in DeFi protocols are commonly exploited. The likelihood factors include:
- Fee manager keys could be compromised through phishing, insider threats, or technical vulnerabilities
- No additional authorization barriers (multi-sig, timelocks) increase exploit probability  
- The function will likely be used legitimately for edge cases, normalizing its usage
- Users cannot easily monitor when their protections are disabled


## Proof of Concept
1. Fee manager calls `turnOffHealthCheck(BOLD_VAULT, MALICIOUS_STRATEGY)`
2. Malicious strategy executes trades that intentionally lose 90% of vault funds
3. Strategy calls `report()` with massive loss amount
4. Health check bypass allows the loss to be processed without reverting
5. Vault users lose 90% of their deposits
6. `skipHealthCheck` flag is reset, hiding evidence of the bypass

// Mock contracts for testing
contract MockVault {
    mapping(address => uint256) public strategies;
    uint256 public totalSupply = 1000e18;
    uint256 public totalAssets = 1000e18;
    
    function setStrategyDebt(address strategy, uint256 debt) external {
        strategies[strategy] = debt;
    }
    
    function setTotalAssets(uint256 assets) external {
        totalAssets = assets;
    }
}

contract MockFeeManager {
    struct Fee {
        bool custom;
        uint256 maxLoss;
        uint256 maxGain;
    }
    
    uint256 public constant MAX_BPS = 10000;
    
    mapping(address => bool) public vaults;
    mapping(address => mapping(address => bool)) public skipHealthCheck;
    mapping(address => Fee) public customConfig;
    Fee public defaultConfig;
    
    address public feeManager;
    
    modifier onlyFeeManager() {
        require(msg.sender == feeManager, "not fee manager");
        _;
    }
    
    modifier onlyAddedVaults() {
        require(vaults[msg.sender], "vault not added");
        _;
    }
    
    constructor() {
        feeManager = msg.sender;
        defaultConfig = Fee({
            custom: false,
            maxLoss: 1000, // 10% max loss
            maxGain: 2000  // 20% max gain
        });
    }
    
    function addVault(address vault) external {
        vaults[vault] = true;
    }
    
    function turnOffHealthCheck(address vault, address strategy) external onlyFeeManager {
        require(vaults[vault], "vault not added");
        skipHealthCheck[vault][strategy] = true;
    }
    
    function report(
        address strategy,
        uint256 gain,
        uint256 loss
    ) public onlyAddedVaults returns (uint256 totalFees) {
        Fee memory fee = customConfig[msg.sender];
        
        if (!fee.custom) {
            fee = defaultConfig;
        }
        
        MockVault vault = MockVault(msg.sender);
        (uint256 current_debt) = vault.strategies(strategy);
        
        if (gain > 0) {
            if (skipHealthCheck[msg.sender][strategy]) {
                skipHealthCheck[msg.sender][strategy] = false;
            } else if (fee.maxGain > 0) {
                require(gain <= (current_debt * fee.maxGain) / MAX_BPS, "too much gain");
            }
            totalFees = gain;
        } else if (loss > 0) {
            if (skipHealthCheck[msg.sender][strategy]) {
                skipHealthCheck[msg.sender][strategy] = false;
            } else if (fee.maxLoss < MAX_BPS) {
                require(loss <= (current_debt * fee.maxLoss) / MAX_BPS, "too much loss");
            }
        }
        
        return totalFees;
    }
}

## Recommendation
Implement time delays and transparency for health check bypasses:

```diff
+ mapping(address => mapping(address => uint256)) public healthCheckBypassTime;
+ uint256 public constant BYPASS_DELAY = 24 hours;
+ 
+ event HealthCheckBypassScheduled(address indexed vault, address indexed strategy, uint256 executeTime);
+ event HealthCheckBypassed(address indexed vault, address indexed strategy);

function turnOffHealthCheck(address vault, address strategy) external virtual onlyFeeManager {
    require(vaults[vault], "vault not added");
    
-   skipHealthCheck[vault][strategy] = true;
+   uint256 executeTime = block.timestamp + BYPASS_DELAY;
+   healthCheckBypassTime[vault][strategy] = executeTime;
+   emit HealthCheckBypassScheduled(vault, strategy, executeTime);
}

+ function executeHealthCheckBypass(address vault, address strategy) external {
+     require(healthCheckBypassTime[vault][strategy] != 0, "bypass not scheduled");
+     require(block.timestamp >= healthCheckBypassTime[vault][strategy], "delay not elapsed");
+     
+     skipHealthCheck[vault][strategy] = true;
+     healthCheckBypassTime[vault][strategy] = 0;
+     emit HealthCheckBypassed(vault, strategy);
+ }
```
