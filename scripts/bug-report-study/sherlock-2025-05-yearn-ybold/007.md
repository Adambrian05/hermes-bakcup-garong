# Centralization Risk in Strategy.sol

### Summary

The strategy contains a hardcoded auction factory address (`AUCTION_FACTORY`) which introduces centralization risk. If the factory contract is compromised or needs to be upgraded, all strategies would be affected. This single point of failure could lead to system-wide issues if the factory is compromised.


### Root Cause

In `Strategy.sol`, the auction factory address is hardcoded as a constant:

https://github.com/sherlock-audit/2025-05-yearn-ybold/blob/88a3b89c05fd6e59ec130c79659195cd05d872af/yv3-liquityv2-sp-strategy/src/Strategy.sol#L66

`
IAuctionFactory public constant AUCTION_FACTORY = IAuctionFactory(0xCfA510188884F199fcC6e750764FAAbE6e56ec40);
`


### Internal Pre-conditions

- Strategy must be deployed with the hardcoded factory address.
- Factory contract must be trusted
- Factory contract must be immutable.

### External Pre-conditions

1. The factory contract must be deployed and verified.

2. The factory contract must be accessible.

3. The factory contract must maintain expected functionality.

### Attack Path

1. The attacker compromises the factory contract.

2. All strategies using the factory become vulnerable.

3. An attacker can:

- Deploy malicious auction contracts.

- Manipulate auction parameters.

- Extract value from all strategies.

### Impact

System-wide risk if the protocol is compromised
Potential loss of funds across all strategies

contract StrategyCentralizationTest is Test {
    LiquityV2SPStrategy public strategy;
    MockPriceFeed public priceFeed;
    MockStabilityPool public stabilityPool;
    MockAuction public auction;
    MockERC20 public asset;
    MockERC20 public collateral;
    
    address public owner;
    address public maliciousOwner;
    address public user;
    
    uint256 constant INITIAL_COLLATERAL = 100 ether;
    uint256 constant INITIAL_ASSET = 1000 ether;
    
    function setUp() public {
        // Setup accounts
        owner = makeAddr("owner");
        maliciousOwner = makeAddr("maliciousOwner");
        user = makeAddr("user");
        
        // Deploy mocks
        priceFeed = new MockPriceFeed();
        stabilityPool = new MockStabilityPool();
        auction = new MockAuction();
        asset = new MockERC20("BOLD", "BOLD", 18);
        collateral = new MockERC20("WETH", "WETH", 18);
        
        // Deploy strategy
        vm.startPrank(owner);
        strategy = new LiquityV2SPStrategy(
            address(stabilityPool),
            address(asset),
            "Strategy"
        );
        
        // Setup strategy
        strategy.setPriceFeed(address(priceFeed));
        strategy.setAuction(address(auction));
        strategy.setBufferPercentage(1100); // 110%
        vm.stopPrank();
        
        // Setup initial state
        stabilityPool.setCollateralGain(address(strategy), INITIAL_COLLATERAL);
        asset.mint(address(strategy), INITIAL_ASSET);
    }

    function testMaliciousOwnerCanDrainFunds() public {
        // 1. Record initial balances
        uint256 initialAssetBalance = asset.balanceOf(address(strategy));
        uint256 initialCollateralBalance = collateral.balanceOf(address(strategy));
        
        console.log("Initial asset balance:", initialAssetBalance);
        console.log("Initial collateral balance:", initialCollateralBalance);
        
        // 2. Simulate owner becoming malicious
        vm.startPrank(owner);
        
        // 3. Malicious owner can:
        // a. Set buffer percentage to 0
        strategy.setBufferPercentage(0);
        console.log("Buffer percentage set to 0");
        
        // b. Set malicious price feed
        MockPriceFeed maliciousPriceFeed = new MockPriceFeed();
        maliciousPriceFeed.setPrice(0); // Set price to 0
        strategy.setPriceFeed(address(maliciousPriceFeed));
        console.log("Malicious price feed set");
        
        // c. Set malicious auction contract
        MockAuction maliciousAuction = new MockAuction();
        strategy.setAuction(address(maliciousAuction));
        console.log("Malicious auction contract set");
        
        // d. Trigger tend with manipulated state
        strategy.tend();
        vm.stopPrank();
        
        // 4. Verify funds were drained
        uint256 finalAssetBalance = asset.balanceOf(address(strategy));
        uint256 finalCollateralBalance = collateral.balanceOf(address(strategy));
        
        console.log("Final asset balance:", finalAssetBalance);
        console.log("Final collateral balance:", finalCollateralBalance);
        
        assertEq(finalAssetBalance, 0, "All assets were drained");
        assertEq(finalCollateralBalance, 0, "All collateral was drained");
    }
    
    function testMaliciousOwnerCanManipulateAuction() public {
        // 1. Setup auction state
        uint256 auctionAmount = 50 ether;
        stabilityPool.setCollateralGain(address(strategy), auctionAmount);
        
        // 2. Record initial state
        uint256 initialAuctionPrice = auction.startingPrice();
        console.log("Initial auction price:", initialAuctionPrice);
        
        // 3. Malicious owner manipulates auction
        vm.startPrank(owner);
        
        // a. Set extremely low buffer percentage
        strategy.setBufferPercentage(100); // 1%
        console.log("Buffer percentage set to 1%");
        
        // b. Set manipulated price feed
        priceFeed.setPrice(1 ether); // Set extremely low price
        console.log("Price feed set to 1 ETH");
        
        // c. Trigger tend
        strategy.tend();
        vm.stopPrank();
        
        // 4. Verify auction was manipulated
        uint256 finalAuctionPrice = auction.startingPrice();
        uint256 expectedPrice = (auctionAmount * 1 ether * 100) / 1e36; // 1% buffer
        
        console.log("Final auction price:", finalAuctionPrice);
        console.log("Expected price with manipulation:", expectedPrice);
        
        assertEq(finalAuctionPrice, expectedPrice, "Auction price was manipulated");
        assertLt(finalAuctionPrice, initialAuctionPrice, "Auction price was reduced");
    }
    
    function testMaliciousOwnerCanBlockWithdrawals() public {
        // 1. Setup withdrawal state
        uint256 withdrawAmount = 10 ether;
        asset.mint(address(strategy), withdrawAmount);
        
        // 2. Record initial state
        uint256 initialAssetBalance = asset.balanceOf(address(strategy));
        console.log("Initial asset balance:", initialAssetBalance);
        
        // 3. Malicious owner blocks withdrawals
        vm.startPrank(owner);
        
        // a. Set malicious price feed that always reverts
        MockPriceFeed revertingPriceFeed = new MockPriceFeed();
        revertingPriceFeed.setShouldRevert(true);
        strategy.setPriceFeed(address(revertingPriceFeed));
        console.log("Reverting price feed set");
        
        // b. Attempt withdrawal (should revert)
        vm.expectRevert("Price feed reverted");
        strategy.withdraw(withdrawAmount);
        vm.stopPrank();
        
        // 4. Verify withdrawal was blocked
        uint256 finalAssetBalance = asset.balanceOf(address(strategy));
        console.log("Final asset balance:", finalAssetBalance);
        
        assertEq(finalAssetBalance, initialAssetBalance, "Withdrawal was blocked");
    }
}

contract MockPriceFeed {
    uint256 public price;
    bool public shouldRevert;
    
    function setPrice(uint256 _price) external {
        price = _price;
    }
    
    function setShouldRevert(bool _shouldRevert) external {
        shouldRevert = _shouldRevert;
    }
    
    function fetchPrice() external view returns (uint256, bool) {
        if (shouldRevert) revert("Price feed reverted");
        return (price, false);
    }
}

contract MockStabilityPool {
    mapping(address => uint256) public collateralGains;
    
    function setCollateralGain(address _user, uint256 _amount) external {
        collateralGains[_user] = _amount;
    }
    
    function getDepositorCollGain(address _user) external view returns (uint256) {
        return collateralGains[_user];
    }
}

contract MockAuction {
    uint256 public startingPrice;
    
    function setStartingPrice(uint256 _price) external {
        startingPrice = _price;

}
