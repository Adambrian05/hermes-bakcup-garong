# H-02: LV2SPStakerStrategy – Mis-routed Reward Claiming Yields Zero Harvest

### Summary

`LV2SPStakerStrategy` calls the underlying vault’s **Accountant** to distribute rewards by invoking `accountant.distribute(address(asset))`, but passes the vault share token address. Because the accountant holds only the vault’s underlying asset (e.g., DAI), it always distributes zero share tokens, so the strategy never receives any active yield.

### Root Cause

- The constructor’s `asset` parameter is the Yearn vault **share** token (e.g., `yvDAI`).  
- Strategy code:  
  ```solidity
  IAccountant ACCOUNTANT = IAccountant(IVault(asset).accountant());
  ACCOUNTANT.distribute(asset);
  ```  
- The vault’s accountant tracks underlying asset fees, so `balanceOf(accountant, asset)` is zero → no yield to strategy.

### Internal Pre-conditions

Strategy deployed with a valid Yearn share token. 

### External Pre-conditions

Keeper or any caller invokes `harvest()`.

### Attack Path

1. Keeper calls `strategy.harvest()`.  
2. Strategy calls `ACCOUNTANT.distribute(yvToken)`.  
3. Accountant transfers `0` shares → strategy’s balance unchanged.  
4. Users expect active harvesting but get only passive share accrual.

### Impact

- **Zero Active Yield:** Performance fees and rewards are never claimed.  
- **User Deception:** Silent failure of core functionality.

### PoC

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.23;

import "./Staker.sol";           // LV2SPStakerStrategy
import "./interfaces/IAccountant.sol";
import "@openzeppelin/contracts/token/ERC20/ERC20.sol";
import "@openzeppelin/contracts/token/ERC20/utils/SafeERC20.sol";

/// @notice Mock ERC20 to represent Yearn share token
contract MockERC20 is ERC20 {
    constructor() ERC20("yvMock", "yvM") {}
    function mint(address to, uint256 amt) external { _mint(to, amt); }
}

/// @notice Mock Accountant that tracks distributions of the passed token
contract MockAccountant is IAccountant {
    using SafeERC20 for ERC20;
    address public override feeRecipient = address(0xFEFE);
    mapping(address => uint256) public distributed;

    function distribute(address token) external override {
        uint256 bal = ERC20(token).balanceOf(address(this));
        if (bal > 0) ERC20(token).safeTransfer(feeRecipient, bal);
        distributed[token] += bal;
    }
    function report(address, uint256, uint256) external pure override returns (uint256, uint256) { return (0,0); }
}

/// @notice Minimal Vault stub linking to our MockAccountant
contract MockVault {
    MockAccountant public acct;
    constructor(MockAccountant _acct) { acct = _acct; }
    function accountant() external view returns (address) { return address(acct); }
}

/// @notice Demonstrates the zero-harvest bug
contract PoC_H02 {
    LV2SPStakerStrategy public strat;
    MockERC20 public yvToken;
    MockAccountant public acct;

    constructor() {
        yvToken = new MockERC20();
        acct   = new MockAccountant();
        address vaultShare = address(yvToken); 
        strat = new LV2SPStakerStrategy(vaultShare, "TestStrat");

        strat.setKeeper(address(this));
        strat.setPerformanceFeeRecipient(address(this));
        strat.setEmergencyAdmin(address(this));
        strat.setPendingManagement(address(this));
        strat.acceptManagement();

        yvToken.mint(address(strat), 100 ether);
        yvToken.mint(address(acct),   5 ether);
    }

    function runHarvest() external {
        uint256 before = yvToken.balanceOf(address(strat));
        strat.harvest();
        uint256 after = yvToken.balanceOf(address(strat));

        require(after == before, "Strategy unexpectedly received share tokens");
        require(acct.distributed(address(yvToken)) == 5 ether, "Accountant did not distribute correctly");
    }
}
```
