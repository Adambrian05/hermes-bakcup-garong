# GatewaySend incorrectly decodes swapData

### Summary

[GatewaySend.sol:decodePackedData()](https://github.com/sherlock-audit/2025-05-dodo-cross-chain-dex/blob/main/omni-chain-contracts/contracts/GatewaySend.sol#L143) incorrectly slices `swapData` starting from byte 40, omitting `tokenA` and `tokenB`, which are part of the expected `MixSwapParams`.

### Root Cause

[MixSwapParams](https://github.com/sherlock-audit/2025-05-dodo-cross-chain-dex/blob/main/omni-chain-contracts/contracts/libraries/SwapDataHelperLib.sol#L6) contains following fields:
```solidity
struct MixSwapParams {
    address fromToken; //@audit focus on this field
    address toToken;     //@audit and this
    uint256 fromTokenAmount;
    uint256 expReturnAmount;
    uint256 minReturnAmount;
    address[] mixAdapters;
    address[] mixPairs;
    address[] assetTo;
    uint256 directions;
    bytes[] moreInfo;
    bytes feeData;
    uint256 deadline;
}
```
[GatewayCrossChain.sol:_handleEvmOrSolanaWithdraw()](https://github.com/sherlock-audit/2025-05-dodo-cross-chain-dex/blob/main/omni-chain-contracts/contracts/GatewayCrossChain.sol#L391) and [GatewayTransferNative.sol:_handleEvmOrSolanaWithdraw()](https://github.com/sherlock-audit/2025-05-dodo-cross-chain-dex/blob/main/omni-chain-contracts/contracts/GatewayTransferNative.sol#L479) encodes message payload like this:
```solidity
bytes memory data = SwapDataHelperLib.buildOutputMessage(
    externalId, 
    outputAmount - gasFee, 
    decoded.receiver, 
    decoded.swapDataB  //@audit @note MixSwapParams
);
```
This message decodes in [GatewaySend.sol:onCall()](https://github.com/sherlock-audit/2025-05-dodo-cross-chain-dex/blob/main/omni-chain-contracts/contracts/GatewaySend.sol#L341):
```solidity
function onCall(
    MessageContext calldata /*context*/,
    bytes calldata message
) external payable onlyGateway returns (bytes4) {
    (
        bytes32 externalId, 
        uint256 amount, 
        bytes calldata recipient, 
        address fromToken, 
        address toToken, 
        bytes calldata swapData 
    ) = decodePackedMessage(message);
    ...
    if(fromToken == toToken) {
        outputAmount = amount;
    } else {
        //@audit @note swap here
        outputAmount = _doMixSwap(swapData);
    }
    ...
}
```
When calling `decodePackedMessage()` contract separately decodes `swapData` in `decodePackedData()`:
```solidity
function decodePackedMessage(
    bytes calldata message
) public pure returns (
    bytes32 externalId, 
    uint256 outputAmount, 
    bytes calldata receiver, 
    address fromToken, 
    address toToken, 
    bytes calldata swapDataB
) {
    uint16 receiverLen;
    uint16 crossChainDataLen;
    bytes calldata crossChainData;

    assembly {
        externalId := calldataload(message.offset) // first 32 bytes
        outputAmount := calldataload(add(message.offset, 32)) // next 32 bytes
        receiverLen := shr(240, calldataload(add(message.offset, 64))) // 2 bytes
        crossChainDataLen := shr(240, calldataload(add(message.offset, 66))) // 2 bytes
    }

    uint offset = 68; // starting point of receiver
    receiver = message[offset : offset + receiverLen];
    offset += receiverLen;
    crossChainData = message[offset : offset + crossChainDataLen];
    //@audit crossChainData is already swapData, so we take fromToken and toToken
    (fromToken, toToken, swapDataB) = decodePackedData(crossChainData);
}
```
But `decodePackedData()` will incorrectly return `swapDataB` without `fromToken` and `toToken` fields:
```solidity
function decodePackedData(bytes calldata data) internal pure returns (
    address tokenA,
    address tokenB,
    bytes calldata swapDataB
) {
    assembly {
        tokenA := shr(96, calldataload(data.offset))
        tokenB := shr(96, calldataload(add(data.offset, 20)))
    }

    if (data.length > 40) {
        //@audit @note first 40 bytes is tokenA and tokenB, which should be in swapData too
        swapDataB = data[40:];
    } else {
        swapDataB = data[0:0]; // empty slice
    }
}
```
This will create DoS for `GatewaySend.sol:onCall()` when user will try to swap tokens.

### PoC

To run PoC make `decodePackedMessage()` function in `GatewaySend.sol` contract `public`.
Then import `SwapDataHelperLib` in `test/GatewaySend.t.sol`:
```solidity
import {SwapDataHelperLib} from "../contracts/libraries/SwapDataHelperLib.sol";
```
And add this test:
```solidity
function test_decodePackedDataIncorrect() public {
    address someAddrReceiver = address(0xaddee);
    address someFromToken = address(0xfce);
    address someToToken = address(0xdce);
    bytes memory receiver = abi.encodePacked(someAddrReceiver);
    bytes memory fromToken = abi.encodePacked(someFromToken);
    bytes memory toToken = abi.encodePacked(someToToken);
    bytes32 someExternalId = keccak256(abi.encodePacked(address(this), someAddrReceiver, uint256(1), block.timestamp));
    uint256 someOutputAmount = uint256(100);
    bytes memory swapDataB = encodeCompressedMixSwapParams(
        someFromToken,
        someToToken,
        98995986959878634903,
        0,
        0,
        new address[](1),
        new address[](1),
        new address[](1),
        0,
        new bytes[](1),
        abi.encode(address(0), 0),
        block.timestamp + 600
    );
    //@audit mock data build
    bytes memory message = SwapDataHelperLib.buildOutputMessage(
        someExternalId, someOutputAmount, receiver, swapDataB);

    (
        bytes32 decodedExternalId,
        uint256 decodedOutputAmount,
        bytes memory decodedReceiver,
        address decodedFromToken,
        address decodedToToken,

    ) = gatewaySendA.decodePackedMessage(
        message
    );

    assertEq(someExternalId, decodedExternalId);
    assertEq(someOutputAmount, decodedOutputAmount);

    assertEq(someFromToken, decodedFromToken);
    assertEq(someToToken, decodedToToken);

    assertNotEq(swapDataB.length, decodedSwapDataB.length);
}
```

### Mitigation

Do not slice data in `decodePackedData()`:
```diff
function decodePackedData(bytes calldata data) internal pure returns (
    address tokenA,
    address tokenB,
    bytes calldata swapDataB
) {
    assembly {
        tokenA := shr(96, calldataload(data.offset))
        tokenB := shr(96, calldataload(add(data.offset, 20)))
    }

    if (data.length > 40) {
-        swapDataB = data[40:];
+        swapDataB = data;
    } else {
        swapDataB = data[0:0]; // empty slice
    }
}
```
