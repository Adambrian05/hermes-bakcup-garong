# Potential Financial Loss from Unchecked Swap Output in _doMixSwap

## Description
The function onCall will be affected with an issue when it calls  _doMixSwap.
The problem arises in the function _doMixSwap, it does not verify that the outputAmount meets params.minReturnAmount. This allows swaps to complete with lower than expected output, exposing users to slippage and potential financial loss

Impact
Unexpected financial loss

Excessive slippage beyond what they explicitly allowed.

        symbol = _symbol;
        decimals = _decimals;

        balanceOf[to] += amount;
        totalSupply += amount;
    }

    function approve(address spender, uint256 amount) external returns (bool) {
        allowance[msg.sender][spender] = amount;
        return true;

        balanceOf[msg.sender] -= amount;
        balanceOf[to] += amount;
        return true;
    }

        require(allowance[from][msg.sender] >= amount, "Insufficient allowance");
        allowance[from][msg.sender] -= amount;
        balanceOf[from] -= amount;
        balanceOf[to] += amount;
        return true;
    }
}

struct MixSwapParams {
    address fromToken;
    address toToken;
    uint256 fromTokenAmount;
    uint256 expReturnAmount;
    uint256 minReturnAmount;
    address[] mixAdapters;
    address[] mixPairs;

// Contract containing _doMixSwap
contract MixSwapWrapper {
    address public immutable DODORouteProxy;

    constructor(address _routeProxy) {
        DODORouteProxy = _routeProxy;
    }

    function _doMixSwap(
        bytes memory swapData,
        uint256 amount,
        MixSwapParams memory params
    ) public returns (uint256 outputAmount) {
        if (swapData.length == 0) {
            return amount;
        }

        console.log("Approving DODORouteProxy:", DODORouteProxy, "for amount:", amount);
        IZRC20(params.fromToken).approve(DODORouteProxy, amount);
        return IDODORouteProxy(DODORouteProxy).mixSwap(
            params.fromToken,
            params.toToken,
            params.fromTokenAmount,
            params.expReturnAmount,
            params.minReturnAmount,
            params.mixAdapters,
            params.mixPairs,
            params.assetTo,
            params.directions,
            params.moreInfo,
            params.feeData,
            params.deadline
        );
    }

    function testDoMixSwap(
        bytes memory swapData,
        uint256 amount,
        MixSwapParams memory params
    ) external returns (uint256 outputAmount) {
        return _doMixSwap(swapData, amount, params);
    }
}

// Test contract
contract DoMixSwapSlippageTest is Test {
    ERC20Mock fromToken;
    ERC20Mock toToken;
    MixSwapWrapper wrapper;
    DODORouteProxyMock routeProxy;
    address user = address(0xA0);

    function setUp() public {
        // Deploy token mocks
        fromToken = new ERC20Mock("FromToken", "FTK", 18);
        toToken = new ERC20Mock("ToToken", "TTK", 18);

        // Deploy route proxy
        routeProxy = new DODORouteProxyMock();

        // Deploy wrapper with route proxy address
        wrapper = new MixSwapWrapper(address(routeProxy));

        // Mint tokens to wrapper and route proxy
        fromToken.mint(address(wrapper), 1000 ether);
        toToken.mint(address(routeProxy), 1000 ether);

        // Reset route proxy state
        routeProxy.setNextOutputAmount(0);
    }

    function initParams() internal view returns (MixSwapParams memory params) {
        params.fromToken = address(fromToken);
        params.toToken = address(toToken);
        params.fromTokenAmount = 1 ether;
        params.expReturnAmount = 1 ether;
        params.minReturnAmount = 0.9 ether;
        params.mixAdapters = new address[](0);
        params.mixPairs = new address[](0);
        params.assetTo = address(wrapper);
        params.directions = 0;
        params.moreInfo = "";
        params.feeData = "";
        params.deadline = block.timestamp + 300;
    }

    function test_lowOutputAllowed_exposesSlippage() public {
        MixSwapParams memory params = initParams();
        params.fromTokenAmount = 1 ether;
        params.minReturnAmount = 0.9 ether; // Expect at least 0.9 ether output
        uint256 amount = 1 ether;
        bytes memory swapData = "0x1234";

        // Set mock output below minReturnAmount
        routeProxy.setNextOutputAmount(0.5 ether); // Output only 0.5 ether
        console.log("Before mixSwap: nextOutputAmount:", routeProxy.nextOutputAmount());

        // Record initial state
        uint256 wrapperFromBalanceBefore = fromToken.balanceOf(address(wrapper));
        uint256 wrapperToBalanceBefore = toToken.balanceOf(address(wrapper));

        // Call mixSwap (should not revert despite low output)
        uint256 outputAmount = wrapper.testDoMixSwap(swapData, amount, params);

        // Verify results
        assertEq(outputAmount, 0.5 ether, "Incorrect output amount");
        assertLt(outputAmount, params.minReturnAmount, "Output should be less than minReturnAmount");
        assertEq(
            fromToken.balanceOf(address(wrapper)),
            wrapperFromBalanceBefore - params.fromTokenAmount,
            "Incorrect fromToken balance"
        );
        assertEq(
            toToken.balanceOf(address(wrapper)),
            wrapperToBalanceBefore + outputAmount,
            "Incorrect toToken balance"
        );
        assertEq(fromToken.balanceOf(user), 0, "User received tokens unexpectedly");
    }
}

## Recommendation
Add a manual check after the swap to ensure outputAmount >= params.minReturnAmount:
```solidity
outputAmount = IDODORouteProxy(DODORouteProxy).mixSwap(
    params.fromToken,
    params.toToken,
    params.fromTokenAmount,
    params.expReturnAmount,
    params.minReturnAmount,
    params.mixAdapters,
    params.mixPairs,
    params.assetTo,
    params.directions,
    params.moreInfo,
    params.feeData,
    params.deadline
);

require(
    outputAmount >= params.minReturnAmount,
    "MixSwap: INSUFFICIENT_OUTPUT_AMOUNT"
);

```
Why This Works
Ensures the swap result meets user-defined slippage protection, regardless of DODO's internal enforcement.

Prevents users from unknowingly receiving fewer tokens than expected.

Adds an extra layer of safety against front-running or unexpected market changes.
