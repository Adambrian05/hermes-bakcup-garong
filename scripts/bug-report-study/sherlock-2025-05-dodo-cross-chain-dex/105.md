# ETH Amount Inconsistency Bug in depositAndCall

### Summary



In the second overload of depositAndCall, there's a critical inconsistency in how ETH amounts are handled:


function depositAndCall(
    address targetContract,
    uint256 amount,           //  User specifies amount
    address asset,
    uint32 dstChainId,
    bytes calldata payload
) public payable {

    // ... setup code ...
    
    if (isETH) {
        require(msg.value >= amount, "INSUFFICIENT AMOUNT: ETH NOT ENOUGH");
        _handleETHDeposit(
            targetContract, 
            msg.value,            //  BUG: Uses msg.value instead of amount
            message, 
            revertOptions
        );
    } else {
        // ERC20 path correctly uses 'amount'
        _handleERC20Deposit(targetContract, amount, asset, message, revertOptions);
    }
}






Problem Analysis

What Should Happen:


User calls depositAndCall(target, 1000, ETH_ADDRESS, chainId, payload)
User sends msg.value = 1000 ETH
Contract should forward exactly 1000 ETH to the target




What Actually Happens:

User calls depositAndCall(target, 1000, ETH_ADDRESS, chainId, payload)
User sends msg.value = 1500 ETH (maybe by mistake)
Contract forwards 1500 ETH instead of 1000 ETH
User loses extra 500 ETH




Real-World Impact


Scenario 1: User Overpayment

// User intends to send 1 ETH cross-chain

depositAndCall(target, 1 ether, ETH_ADDRESS, 7000, payload);

// But accidentally sends 2 ETH as msg.value
// Result: 2 ETH is forwarded, user loses 1 ETH




Scenario 2: Integration Issues

// Frontend calculates exact amount needed

const amount = calculateSwapAmount(); // 0.5 ETH
const gasBuffer = 0.1; // Safety buffer

// Sends transaction with buffer
await contract.depositAndCall(
    target, 
    amount,                    // 0.5 ETH
    ETH_ADDRESS, 
    chainId, 
    payload,
    { value: amount + gasBuffer } // 0.6 ETH sent
);

// BUG: 0.6 ETH forwarded instead of 0.5 ETH
// User loses 0.1 ETH buffer


Scenario 3: Contract Integration

// Another contract calls this function

contract Integrator {
    function bridgeExactAmount() external payable {
        uint256 exactAmount = 1 ether;
        
        // Contract sends exact amount but adds some buffer
        gatewaySend.depositAndCall{value: exactAmount + 0.1 ether}(
            target,
            exactAmount,  // Should only send 1 ETH
            ETH_ADDRESS,
            chainId,
            payload
        );
        // BUG: 1.1 ETH sent instead of 1 ETH
    }
}








Why This Bug is Critical


Silent Fund Loss: Users lose excess ETH without realizing it
Parameter Meaningless: The amount parameter becomes ignored for ETH
Integration Confusion: Other contracts can't predict behavior
No Recovery: Lost ETH goes to the target contract permanently
Inconsistent Behavior: ETH and ERC20 paths behave differently

### Mitigation

function depositAndCall(
    address targetContract,
    uint256 amount,
    address asset,
    uint32 dstChainId,
    bytes calldata payload
) public payable {
    globalNonce++;
    bytes32 externalId = _calcExternalId(msg.sender);
    bool isETH = (asset == _ETH_ADDRESS_);
    bytes memory message = concatBytes(externalId, payload);

    RevertOptions memory revertOptions = RevertOptions({
        revertAddress: address(this),
        callOnRevert: true,
        abortAddress: targetContract,
        revertMessage: bytes.concat(externalId, bytes20(msg.sender)),
        onRevertGasLimit: gasLimit
    });

    if (isETH) {
        require(msg.value >= amount, "INSUFFICIENT AMOUNT: ETH NOT ENOUGH");
        _handleETHDeposit(
            targetContract, 
            amount,              //  FIXED: Use amount instead of msg.value
            message, 
            revertOptions
        );
        
        // Handle excess ETH refund
        if (msg.value > amount) {
            payable(msg.sender).transfer(msg.value - amount);
        }
    } else {
        require(
            IERC20(asset).transferFrom(msg.sender, address(this), amount),
            "INSUFFICIENT AMOUNT: ERC20 TRANSFER FROM FAILED"
        );
        _handleERC20Deposit(targetContract, amount, asset, message, revertOptions);
    }

    emit EddyCrossChainSend(
        externalId,
        dstChainId,
        asset,
        asset,
        amount,        //  Correct: amount, not msg.value
        amount,        //  Correct: amount, not msg.value  
        msg.sender,
        message
    );
}
