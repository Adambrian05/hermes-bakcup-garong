# withdraw Function Inaccessible to Users Due to internal Visibility


## Description
In contract GatewayCrossChain,
The function withdraw is declared as internal which prevents users from withdrawing their funds

Why is this a problem?
Users who want to withdraw their funds cannot call this function directly, because external calls from outside the contract are disallowed for internal functions.

This effectively blocks users from initiating withdrawals themselves, which defeats the purpose of having a withdraw mechanism exposed.

## Proof of Concept
```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import {Test} from "forge-std/Test.sol";
import {GatewayEVMMock} from "../contracts/mocks/GatewayEVMMock.sol";
import {ERC20Mock} from "../contracts/mocks/ERC20Mock.sol";



contract WithdrawInternalMock {
    mapping(address => uint256) public balances;

    constructor() {
        // For testing, give the user some balance
        balances[msg.sender] = 100 ether;
    }

    // Internal: inaccessible from external callers
    function withdraw(uint256 amount) internal {
        require(balances[msg.sender] >= amount, "Insufficient balance");
        balances[msg.sender] -= amount;
    }

    function attemptWithdraw(uint256 amount) external {
        // Simulate that users can't access withdraw
        revert("Withdraw function not accessible externally");
    }
}

contract WithdrawInternalVisibilityTest is Test {
    WithdrawInternalMock internal withdrawContract;

    function setUp() public  {
        
        withdrawContract = new WithdrawInternalMock();
    }


    function testWithdrawInternalPreventsUserAccess() public {
        uint256 initialBalance = withdrawContract.balances(address(this));
        assertEq(initialBalance, 100 ether, "Initial balance should be 100 ether");

        // Try calling internal withdraw (this will fail)
        vm.expectRevert("function selector was not recognized and there's no fallback function");
        (bool success, ) = address(withdrawContract).call(
            abi.encodeWithSignature("withdraw(uint256)", 10 ether)
        );
        assertFalse(success, "Withdraw should fail because it is internal");

        // Balance should remain unchanged
        uint256 finalBalance = withdrawContract.balances(address(this));
        assertEq(finalBalance, initialBalance, "Balance should remain unchanged if withdraw is internal");
    }
}
```

## Recommendation
Change the visibility of withdraw from internal to external or public, so users or other external actors can call it.
