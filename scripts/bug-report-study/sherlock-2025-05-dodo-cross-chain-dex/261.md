# `AccountEncoder::decompressAccounts` does not decode `isWritable` correctly

### Summary

The protocol expects accounts to be encoded in a specific compressed format in order for them to be then decompressed correctly.

The protocol team confirmed that each `Account` entry, must be encoded like:
![ProtocolConfirmation](https://i.postimg.cc/LXtkYdcJ/Screenshot-2025-06-07-at-17-26-14.png)
```solidity
function compressAccounts(bytes32[] memory publicKeys, bool[] memory isWritables) public pure returns (bytes memory out) {
        uint256 len = publicKeys.length;
        require(len == isWritables.length, "Length mismatch");
        require(len < type(uint16).max, "Too many accounts");

        uint256 totalSize = 2 + len * 33;
        out = new bytes(totalSize);

        assembly {
            let ptr := add(out, 32)

            // accounts.length (uint16)
            mstore8(ptr, shr(8, len))
            mstore8(add(ptr, 1), and(len, 0xff))
            ptr := add(ptr, 2)

            for { let i := 0 } lt(i, len) { i := add(i, 1) } {
                let pubkey := mload(add(add(publicKeys, 32), mul(i, 32)))
                let writable := mload(add(add(isWritables, 32), mul(i, 32)))

                mstore(ptr, pubkey)
                ptr := add(ptr, 32)
                mstore8(ptr, iszero(iszero(writable)))
                ptr := add(ptr, 1)
            }
        }
    }
```

The first 2 bytes represent the length of the compressed accounts, how many Accounts do we have.
The pubkeys are 32 bytes long and `isWritable` is 1 byte, as it's a bool.

If we inspect [decompressAccounts](https://github.com/sherlock-audit/2025-05-dodo-cross-chain-dex/blob/main/omni-chain-contracts/contracts/libraries/AccountEncoder.sol#L19-L20) we can see that `isWritable` is always assumed to be 32 bytes long, which is a problem as the code will attempt to read 1 byte from the actual `isWritable` and then 31 bytes from the next pubkey entry.

This dirty reading leads to all `isWrtiable` entires except the last one to be always `true`, since pubkeys can never be 0.

This will lead to incorrectly encoding the Solana accounts and what [ZetaChain is expecting](https://www.zetachain.com/docs/developers/chains/solana/#withdraw-and-call-a-solana-program)

### Root Cause

Wrong Solana accounts `isWritable` decoding

### PoC

```solidity
contract AccountEncodingTest is Test {
    struct SolanaAccount {
        bytes32 publicKey;
        bool isWritable;
    }

    struct Input {
        SolanaAccount[] accounts;
        bytes data;
    }

    function setUp() public {}

    function compressAccounts(bytes32[] memory publicKeys, bool[] memory isWritables) public pure returns (bytes memory out) {
        uint256 len = publicKeys.length;
        require(len == isWritables.length, "Length mismatch");
        require(len < type(uint16).max, "Too many accounts");

        uint256 totalSize = 2 + len * 33;
        out = new bytes(totalSize);

        assembly {
            let ptr := add(out, 32)

            // accounts.length (uint16)
            mstore8(ptr, shr(8, len))
            mstore8(add(ptr, 1), and(len, 0xff))
            ptr := add(ptr, 2)

            for { let i := 0 } lt(i, len) { i := add(i, 1) } {
                let pubkey := mload(add(add(publicKeys, 32), mul(i, 32)))
                let writable := mload(add(add(isWritables, 32), mul(i, 32)))

                mstore(ptr, pubkey)
                ptr := add(ptr, 32)
                mstore8(ptr, iszero(iszero(writable)))
                ptr := add(ptr, 1)
            }
        }
    }

    function decompressAccounts(bytes memory input) internal pure returns (SolanaAccount[] memory accounts) {
        assembly {
            let ptr := add(input, 32)
            
            // Read accounts length (uint16)
            let len := add(shl(8, byte(0, mload(ptr))), byte(1, mload(ptr)))
            ptr := add(ptr, 2)
            
            // Allocate memory for Account[] array
            accounts := mload(0x40)
            mstore(accounts, len)
            let arrData := add(accounts, 32)
            
            // Prepare memory for Account structs
            let freePtr := add(arrData, mul(len, 32))
            
            for { let i := 0 } lt(i, len) { i := add(i, 1) } {
                let acc := freePtr
                freePtr := add(freePtr, 64)
                
                // Load publicKey
                mstore(acc, mload(ptr))
                ptr := add(ptr, 32)
                
                // Load isWritable (as bool)
                mstore(add(acc, 32), iszero(iszero(mload(ptr))))
                ptr := add(ptr, 1)
                
                // Store pointer to struct in the array
                mstore(add(arrData, mul(i, 32)), acc)
            }
            
            mstore(0x40, freePtr)
        }
    }

    function test_AccountEncodingDecoding() public {
        // Create test data for 3 accounts
        bytes32[] memory publicKeys = new bytes32[](3);
        bool[] memory isWritables = new bool[](3);
        
        // All the below entries have `isWritable` set to false,
        // but when decoded, only the last one will return false
        // because of this the test will fail on the second assert
        // between `decoded[i].isWritable` and `isWritables[i]`
        publicKeys[0] = keccak256(abi.encodePacked("account1"));
        isWritables[0] = true;

        publicKeys[1] = keccak256(abi.encodePacked("account2"));
        isWritables[1] = false;

        publicKeys[2] = keccak256(abi.encodePacked("account3"));
        isWritables[2] = false;

        // Encode the accounts
        bytes memory encoded = compressAccounts(publicKeys, isWritables);
        
        // Decode the accounts
        SolanaAccount[] memory decoded = decompressAccounts(encoded);

        // Verify the results
        assertEq(decoded.length, 3, "Wrong number of accounts decoded");

        // Check each account
        for(uint i = 0; i < 3; i++) {
            assertEq(decoded[i].publicKey, publicKeys[i], string(abi.encodePacked("Public key mismatch for account ", vm.toString(i))));
            assertEq(decoded[i].isWritable, isWritables[i], string(abi.encodePacked("isWritable mismatch for account ", vm.toString(i))));
        }
    }
} 
```

### Mitigation

Only read 1 byte for isWritable instead of 32.
