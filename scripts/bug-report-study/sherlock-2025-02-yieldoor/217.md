# Owner can disrupt contract functionality by creating excessive reserves, causing unsustainable gas consumption

### Summary

The lack of a reserve creation limit in `LendingPool.sol` will cause significant disruption in contract functionality as the owner can create an excessive number of reserves, leading to unsustainable gas consumption (e.g., 143M gas for 100 reserves), rendering the contract impractical or inoperable. This violates implicit scalability expectations as a fork of Extra Finance, where pool management is designed to remain efficient  and disrupts core lending operations.


### Root Cause

In `LendingPool.sol' (https://github.com/sherlock-audit/2025-02-yieldoor/blob/main/yieldoor/src/LendingPool.sol#L49-L64) the `initReserve` function allows the owner to create unlimited reserves without restriction, deploying a new `yToken` contract and and increasing the size of the reserves mapping with each call. This unbounded growth leads to excessive gas consumption (e.g., 143M gas for 100 reserves), with no mitigation akin to Extra Finance’s design principles, where pool efficiency is maintained through mechanisms like PFIR.


### Internal Pre-conditions

1. Owner needs to call `initReserve()` repeatedly to create a large number of reserves (e.g., 100 reserves with unique assets).
2. Each reserve requires a unique `asset` address (e.g., distinct ERC20 tokens).



### External Pre-conditions

None – the issue relies solely on internal protocol state and owner actions.



### Attack Path

1. Owner calls `initReserve(address(asset))` 100 times with unique `asset` addresses, creating 100 reserves and deploying 100 `yToken` contracts, significantly increasing the `reserves` mapping size.
2. Each `initReserve` call consumes gas (e.g., ~1.43M per call), totaling 143M gas for 100 reserves, exceeding practical transaction limits (e.g., Ethereum block gas limit of ~30M).
3. Subsequent operations (e.g., `deposit` or `redeem` on any reserve) become prohibitively expensive or infeasible due to cumulative gas costs, disrupting core lending functionality for all users.



### Impact

The owner’s ability to create excessive reserves causes significant disruption in contract functionality, rendering the lending pool impractical or inoperable due to unsustainable gas consumption (e.g., 143M gas for 100 reserves, far exceeding Ethereum’s ~30M block limit). This impacts all users by escalating transaction costs (>0.01% and >$10 in gas fees) and potentially blocking lending operations, violating implicit scalability expectations as a fork of Extra Finance, where pool management is designed to remain efficient.

``` solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.13;

import "forge-std/Test.sol";
import {LendingPool} from "../src/LendingPool.sol";
import {yToken} from "../src/yToken.sol";
import {IERC20} from "@openzeppelin/token/ERC20/IERC20.sol";
import {SafeERC20} from "@openzeppelin/token/ERC20/utils/SafeERC20.sol";

contract MockERC20 is IERC20 {
    using SafeERC20 for IERC20;

    mapping(address => uint256) public override balanceOf;
    mapping(address => mapping(address => uint256)) public override allowance;
    uint256 public totalSupply;

    string public name;
    string public symbol;
    uint8 public decimals = 18;

    constructor(string memory _name, string memory _symbol) {
        name = _name;
        symbol = _symbol;
        balanceOf[msg.sender] = 1000 * 1e18;
        totalSupply = 1000 * 1e18;
    }

    function transfer(
        address to,
        uint256 amount
    ) external override returns (bool) {
        require(balanceOf[msg.sender] >= amount, "Insufficient balance");
        balanceOf[msg.sender] -= amount;
        balanceOf[to] += amount;
        emit Transfer(msg.sender, to, amount);
        return true;
    }

    function approve(
        address spender,
        uint256 amount
    ) external override returns (bool) {
        allowance[msg.sender][spender] = amount;
        emit Approval(msg.sender, spender, amount);
        return true;
    }

    function transferFrom(
        address from,
        address to,
        uint256 amount
    ) external override returns (bool) {
        require(balanceOf[from] >= amount, "Insufficient balance");
        require(
            allowance[from][msg.sender] >= amount,
            "Insufficient allowance"
        );
        balanceOf[from] -= amount;
        balanceOf[to] += amount;
        allowance[from][msg.sender] -= amount;
        emit Transfer(from, to, amount);
        return true;
    }
}

contract ReserveLimitTest is Test {
    LendingPool public lendingPool;
    address public owner = address(this);
    uint256 constant MAX_GAS_THRESHOLD = 15_000_000; // Threshold set to half Ethereum block limit

    function setUp() public {
        lendingPool = new LendingPool();
    }

    function testGasConsumptionWithManyReserves() public {
        uint256 reserveCount = 100; // Testing with 100 reserves
        address[] memory assets = new address[](reserveCount);

        // Create unique tokens as assets
        for (uint256 i = 0; i < reserveCount; i++) {
            string memory name = string(
                abi.encodePacked("Token", vm.toString(i))
            );
            string memory symbol = string(
                abi.encodePacked("TKN", vm.toString(i))
            );
            MockERC20 token = new MockERC20(name, symbol);
            assets[i] = address(token);
        }

        // Measure gas for adding reserves
        uint256 totalGasUsed = 0;
        for (uint256 i = 0; i < reserveCount; i++) {
            uint256 gasBefore = gasleft();
            lendingPool.initReserve(assets[i]);
            uint256 gasAfter = gasleft();
            uint256 gasUsed = gasBefore - gasAfter;
            totalGasUsed += gasUsed;

            // Verify yToken creation
            address yTokenAddr = lendingPool.getYTokenAddress(assets[i]);
            assertTrue(
                yTokenAddr != address(0),
                "yToken should be initialized for each reserve"
            );

            // Log gas usage for debugging
            console.log("Gas used for reserve", i + 1, ":", gasUsed);
        }

        console.log(
            "Total gas used for",
            reserveCount,
            "reserves:",
            totalGasUsed
        );

        // Assert: Total gas should not exceed practical threshold
        assertTrue(
            totalGasUsed < MAX_GAS_THRESHOLD,
            "Total gas consumption exceeds practical threshold"
        );

        // Test deposit functionality with many reserves
        vm.startPrank(owner);
        MockERC20(assets[0]).approve(address(lendingPool), 100 * 1e18);
        lendingPool.deposit(assets[0], 100 * 1e18, owner);
        vm.stopPrank();

        // Verify yToken minting
        address yTokenAddr = lendingPool.getYTokenAddress(assets[0]);
        uint256 yBalance = IERC20(yTokenAddr).balanceOf(owner);
        assertGt(
            yBalance,
            0,
            "Deposit should mint yTokens despite many reserves"
        );
    }
}


```

### Mitigation

To prevent excessive gas consumption and ensure contract scalability, implement a reserve creation limit in `LendingPool.sol`. Add a counter (e.g., `uint256 public reserveCount`) and a maximum limit (e.g., `uint256 public constant MAX_RESERVES = 50`) in the contract. Modify `initReserve` to check `require(reserveCount < MAX_RESERVES, "Maximum reserve limit reached")` before creating a new reserve, incrementing `reserveCount` upon success. This aligns with practical scalability expectations and mitigates the risk of rendering the contract inoperable.
