# First depositor will lose funds due to incorrect share calculation

### Summary

Subtracting a constant value of 1000 from initial shares will cause a loss of funds for first depositor as they will receive fewer shares than their deposit entitles them to, preventing full withdrawal of their initial deposit.

### Root Cause

In [src/Vault.sol#L129-L133](https://github.com/sherlock-audit/2025-02-yieldoor/blob/main/yieldoor/src/Vault.sol#L129-L133) the first deposit share calculation subtracts a constant 1000 from the deposit amount, regardless of token decimals:

```solidity
if (totalSupply == 0) {
    amount0 = depositAmount0;
    amount1 = depositAmount1; 
    shares = (amount0 > amount1 ? amount0 : amount1) - (1000);
}
```


### Internal Pre-conditions

- Vault must have no prior deposits (totalSupply == 0)
- First deposit must be made with a low decimal token (e.g. GUSD with 2 decimals)

### External Pre-conditions

External pre-conditions are absent

### Attack Path

1. First user deposits 20 GUSD (2000 base units) into empty vault
2. User receives only 1000 shares instead of 2000 due to constant 1000 subtraction
3. When withdrawing all shares, user can only recover 10 GUSD (1000 base units)

### Impact

The first depositor suffers a loss proportional to 1000/depositAmount. For low decimal tokens like GUSD (2 decimals), depositing 20 GUSD results in a 50% loss of funds (10 GUSD). The lost funds remain locked in the contract.

```solidity
import {Strategy} from "../src/Strategy.sol";

import {LiquidityAmounts} from "../src/libraries/LiquidityAmounts.sol";
import {TickMath} from "../src/libraries/TickMath.sol";

contract PocTest is Test {
    IUniswapV3Pool pool = IUniswapV3Pool(0x5aA1356999821b533EC5d9f79c23B8cB7c295C61); // GUSD/USDC
    IERC20 gusd = IERC20(0x056Fd409E1d7A124BD7017459dFEa2F387b6d5Cd); // token0 e2
    IERC20 usdc = IERC20(0xA0b86991c6218b36c1d19D4a2e9Eb0cE3606eB48); // token1 e6
    address vault;
    address strategy;
    address rebalancer = address(1001);

        uint256 mainnetFork = vm.createFork('https://go.getblock.io/<ACCESS_TOKEN>');
        vm.selectFork(mainnetFork);

        vm.startPrank(owner);
        vault = address(new Vault(address(gusd), address(usdc)));
        strategy = address(new Strategy(address(pool), address(vault), rebalancer, feeRecipient));
        IVault(vault).setStrategy(strategy);
        
        vm.stopPrank();
    }

    function test_IncorrectWithdrawAmount() public {
        address alice = address(0x0123455);
        address gusdWhale = address(0x5f65f7b609678448494De4C87521CdF6cEf1e932);
        uint256 gusdDepositAmount = 20e2;
        uint256 usdcDepositAmount = 0;

        vm.startPrank(gusdWhale);
        IERC20(gusd).transfer(alice, gusdDepositAmount);
        vm.stopPrank();

        deal(address(usdc), alice, usdcDepositAmount);

        // Alice deposits
        vm.startPrank(alice);

        gusd.approve(vault, type(uint256).max);
        usdc.approve(vault, type(uint256).max);

        (uint256 shares, uint256 amount0In, uint256 amount1In) =

        uint256 expectedShares = (amount0In > amount1In ? amount0In : amount1In); 
        assertEq(shares, expectedShares - 1000); // 1000 is pre-minted tokens to protect against Inflation Attack
        
        assertEq(gusd.balanceOf(alice), 0);
        assertEq(usdc.balanceOf(alice), 0);

        // Try to withdraw a deposit
        uint256 aliceShares = IVault(vault).balanceOf(alice);

        (uint256 withdrawAmount0, uint256 withdrawAmount1) = IVault(vault).withdraw(aliceShares, 0, 0);
        
        console.log("Withdraw Amount 0: %d", withdrawAmount0);
        console.log("Withdraw Amount 1: %d", withdrawAmount1);

        // Assertion failed, gusdDepositAmount - withdrawAmount0 = 1000 
        // [FAIL: token0: incorrect withdraw amount: 1000 != 2000] test_IncorrectWithdrawAmount()
        assertEq(withdrawAmount0, gusdDepositAmount, "token0: incorrect withdraw amount");

        vm.stopPrank();
    }
}
```

### Mitigation

Instead of subtracting a constant value, scale the initial share deduction based on the token decimals
