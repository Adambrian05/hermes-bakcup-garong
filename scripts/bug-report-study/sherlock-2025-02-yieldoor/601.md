# Inadequate Slippage Protection in `Vault::deposit` Leading to Value Loss

### Summary

The [`Vault::deposit`](https://github.com/sherlock-audit/2025-02-yieldoor/blob/main/yieldoor/src/Vault.sol#L48) function has minimal slippage protection that only verifies individual token minimums without considering the ratio between tokens or implementing a deadline parameter. This makes the function vulnerable to sandwich attacks where an attacker can manipulate the pool ratio to force users to deposit with unfavorable token ratios, leading to value loss even when individual token minimums are met.

### Root Cause

The `deposit` function implements basic slippage protection that only checks individual token amounts:

```solidity
function deposit(uint256 amount0, uint256 amount1, uint256 amount0Min, uint256 amount1Min)
    external
    returns (uint256 shares, uint256 depositAmount0, uint256 depositAmount1)
{
    IStrategy(strategy).collectFees();

    (uint256 totalBalance0, uint256 totalBalance1) = IStrategy(strategy).balances();

    (shares, depositAmount0, depositAmount1) = _calcDeposit(totalBalance0, totalBalance1, amount0, amount1);

    IERC20(token0).safeTransferFrom(msg.sender, strategy, depositAmount0);
    IERC20(token1).safeTransferFrom(msg.sender, strategy, depositAmount1);

    if (depositFee > 0) shares -= (shares * depositFee) / 10_000;

    require(shares > 0, "shares cant be 0");
    require(depositAmount0 >= amount0Min, "slippage0");
    require(depositAmount1 >= amount1Min, "slippage1");

    if (totalSupply() == 0) _mint(BURN, 1000);
    _mint(msg.sender, shares);

    return (shares, depositAmount0, depositAmount1);
}
```

The key limitations of this implementation are:

1. It doesn't verify the ratio between token0 and token1, only their individual minimums
2. There's no deadline parameter to prevent stale transactions
3. The actual deposited amounts are calculated based on the existing pool ratio in `_calcDeposit`, which can be manipulated


### Internal Pre-conditions

1. The Vault contract must have existing deposits
2. The market must be volatile or the underlying pool must be susceptible to manipulation

### External Pre-conditions

.

### Attack Path

1. Alice prepares to deposit tokens into the Vault with basic slippage protection
2. Bob identifies Alice's transaction in the mempool
3. Bob front-runs Alice by executing a large swap that changes the ratio of token0/token1 in the pool
4. Alice's transaction executes with the manipulated ratio, which affects the amounts calculated by `_calcDeposit`
5. The individual token minimums are still satisfied, so the transaction succeeds
6. Alice has now deposited with an unfavorable ratio, receiving fewer high-value tokens and more low-value tokens
7. Bob back-runs with a swap to restore the price and profits from the manipulation

### Impact

Users will receive different token ratios than expected when the pool state changes between transaction submission and execution. This can lead to value loss even when transactions meet individual token minimum requirements. In volatile markets or during periods of high gas prices when transactions might remain pending longer, users are particularly vulnerable to this type of attack.

contract SimpleVault is ERC20 {
    address public token0;
    address public token1;
    address public strategy;
    
    uint256 internal strat_balance0;
    uint256 internal strat_balance1;
    
    constructor(address _token0, address _token1) ERC20("Test Vault", "vTEST") {
        token0 = _token0;
        token1 = _token1;
    }
    
    function setStrategyBalances(uint256 _balance0, uint256 _balance1) external {
        strat_balance0 = _balance0;
        strat_balance1 = _balance1;
    }
    
    function balances() public view returns (uint256, uint256) {
        return (strat_balance0, strat_balance1);
    }
    
    function deposit(uint256 amount0, uint256 amount1, uint256 amount0Min, uint256 amount1Min)
        external
        returns (uint256 shares, uint256 depositAmount0, uint256 depositAmount1)
    {
        (uint256 totalBalance0, uint256 totalBalance1) = balances();
        
        (shares, depositAmount0, depositAmount1) = _calcDeposit(
            totalBalance0, totalBalance1, amount0, amount1
        );
        
        IERC20(token0).transferFrom(msg.sender, address(this), depositAmount0);
        IERC20(token1).transferFrom(msg.sender, address(this), depositAmount1);
        
        require(depositAmount0 >= amount0Min, "slippage0");
        require(depositAmount1 >= amount1Min, "slippage1");
        
        strat_balance0 += depositAmount0;
        strat_balance1 += depositAmount1;
        
        _mint(msg.sender, shares);
        
        return (shares, depositAmount0, depositAmount1);
    }
    
    function _calcDeposit(
        uint256 bal0, uint256 bal1, uint256 depositAmount0, uint256 depositAmount1
    ) internal view returns (uint256 shares, uint256 amount0, uint256 amount1) {
        uint256 totalSupply = totalSupply();
        
        if (totalSupply == 0) {
            amount0 = depositAmount0;
            amount1 = depositAmount1;
            shares = amount0 > amount1 ? amount0 : amount1;
        } else if (bal0 == 0) {
            amount1 = depositAmount1;
            shares = amount1 * totalSupply / bal1;
        } else if (bal1 == 0) {
            amount0 = depositAmount0;
            shares = amount0 * totalSupply / bal0;
        } else {
            uint256 cross0 = depositAmount0 * bal1;
            uint256 cross1 = depositAmount1 * bal0;
            uint256 cross = cross0 < cross1 ? cross0 : cross1;
            
            amount0 = (cross + bal1 - 1) / bal1;
            amount1 = (cross + bal0 - 1) / bal0;
            shares = cross * totalSupply / bal0 / bal1;
        }
    }
}

### Mitigation


1. Add a deadline parameter to prevent stale transactions:
   ```solidity
   function deposit(
       uint256 amount0, 
       uint256 amount1, 
       uint256 amount0Min, 
       uint256 amount1Min,
       uint256 deadline
   ) external {
       require(block.timestamp <= deadline, "transaction expired");
   }
   ```

2. Add ratio-based slippage protection:
   ```solidity
   function deposit(...) external {
       uint256 depositRatio = depositAmount0 * 1e18 / depositAmount1;
       uint256 expectedRatio = amount0 * 1e18 / amount1;
       uint256 maxRatioSlippage = 500;
       
       require(
           depositRatio >= expectedRatio * (10000 - maxRatioSlippage) / 10000 &&
           depositRatio <= expectedRatio * (10000 + maxRatioSlippage) / 10000,
           "deposit ratio outside acceptable range"
       );
   }
   ```
