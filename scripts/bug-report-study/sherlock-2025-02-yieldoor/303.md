# Slippage Vulnerability in `openLeveragedPosition` Function

### Summary

Lack of validation for user-provided slippage parameters will cause significant financial loss for the protocol and users as attackers can perform sandwich attacks to extract value through price manipulation.

### Root Cause

In [openLeveragedPosition()](https://github.com/sherlock-audit/2025-02-yieldoor/blob/main/yieldoor/src/Leverager.sol#L103) function, the contract uses the user-provided `amountInMaximum` parameter without validation, allowing users to set unlimited slippage protection by using `type(uint256).max`.

```solidity
  if (a0 > lp.amount0In && up.token0 != lp.denomination) {
                swapParams = abi.decode(lp.swapParams1, (IMainnetRouter.ExactOutputParams));

                address tokenIn = _getTokenIn(swapParams.path);
                require(tokenIn == lp.denomination, "token should be denomination");

                IERC20(tokenIn).forceApprove(swapRouter, swapParams.amountInMaximum);
```

The code only updates `amountOut` but continues to use the user-provided `amountInMaximum` without any validation or sensible limits.

### Attack Path

1. Attacker monitors mempool for transactions calling `openLeveragedPosition()`

2. Attacker identifies a transaction with `high amountInMaximum` in the swap parameters
 
3. Attacker front-runs by executing a large trade to temporarily manipulate the price upward
 
4. Target transaction executes with no slippage protection, paying the manipulated (higher) price
 
5. Attacker back-runs by executing a trade to return the price to normal, profiting from the difference

### Impact

The protocol suffers significant financial losses when executing swaps with unlimited slippage protection. When a sandwich attack occurs, the protocol will pay substantially more than the fair market value for tokens, with the excess amount directly extractable by attackers.

```solidity
contract MockERC20 {
    string public name;
    string public symbol;
    uint8 public decimals;

    mapping(address => uint256) public balanceOf;
    mapping(address => mapping(address => uint256)) public allowance;

    constructor(string memory _name, string memory _symbol, uint8 _decimals) {
        name = _name;
        symbol = _symbol;
        decimals = _decimals;
    }

    function mint(address to, uint256 amount) external {
        balanceOf[to] += amount;
    }

    function approve(address spender, uint256 amount) external returns (bool) {
        allowance[msg.sender][spender] = amount;
        return true;
    }

    function transfer(address to, uint256 amount) external returns (bool) {
        require(balanceOf[msg.sender] >= amount, "Insufficient balance");
        balanceOf[msg.sender] -= amount;
        balanceOf[to] += amount;
        return true;
    }

    function transferFrom(address from, address to, uint256 amount) external returns (bool) {
        require(allowance[from][msg.sender] >= amount, "Insufficient allowance");
        require(balanceOf[from] >= amount, "Insufficient balance");

        allowance[from][msg.sender] -= amount;
        balanceOf[from] -= amount;
        balanceOf[to] += amount;
        return true;
    }
}

// This contract simulates the Uniswap router and the price manipulation
contract MockExchange {
    bool public manipulatedPrice = false;
    uint256 public normalRate = 1e18;     // 1:1 exchange rate normally
    uint256 public manipulatedRate = 3e18; // 3:1 exchange rate when manipulated

    MockERC20 public token0;
    MockERC20 public token1;

    constructor(MockERC20 _token0, MockERC20 _token1) {
        token0 = _token0;
        token1 = _token1;
    }

    function setManipulatedPrice(bool _manipulated) external {
        manipulatedPrice = _manipulated;
    }

    // Simplified swap function that simulates `exactOutput` behavior
    function swap(uint256 amountOut, uint256 amountInMaximum, address recipient) external returns (uint256) {
        // Calculate input amount based on the current rate
        uint256 rate = manipulatedPrice ? manipulatedRate : normalRate;
        uint256 amountIn = (amountOut * rate) / 1e18;

        require(amountIn <= amountInMaximum, "Too much requested");

        token0.transfer(recipient, amountOut);
        token1.transferFrom(msg.sender, address(this), amountIn);

        return amountIn;
    }
}

// This contract simulates the vulnerable part of the Leverager contract
contract MockLeverager {
    MockERC20 public token0;
    MockERC20 public token1;
    MockExchange public exchange;

    constructor(MockERC20 _token0, MockERC20 _token1, MockExchange _exchange) {
        token0 = _token0;
        token1 = _token1;
        exchange = _exchange;
    }

    function swapWithoutProtection(
        uint256 userAmountIn,  // Amount of token0 the user provides
        uint256 totalDesired    // Total amount of token0 desired
    ) external returns (uint256) {
        token0.transferFrom(msg.sender, address(this), userAmountIn);

        uint256 amountNeeded = totalDesired - userAmountIn;

        if (amountNeeded > 0) {
            require(token1.balanceOf(address(this)) >= amountNeeded, "Insufficient token1 in contract");

            // VULNERABILITY: Use unlimited slippage protection
            uint256 amountInMaximum = type(uint256).max;

            token1.approve(address(exchange), amountInMaximum);

            uint256 amountSpent = exchange.swap(amountNeeded, amountInMaximum, msg.sender);

            return amountSpent;
        }

        return 0;
    }
}

contract SlippageVulnerabilityTest is Test {
    MockERC20 token0;
    MockERC20 token1;
    MockExchange exchange;
    MockLeverager leverager;

    address user = address(0xBEEF);
    address attacker = address(0xDEAD);

    function setUp() public {
        token0 = new MockERC20("Token0", "TKN0", 18);
        token1 = new MockERC20("Token1", "TKN1", 18);

        exchange = new MockExchange(token0, token1);
        leverager = new MockLeverager(token0, token1, exchange);

        token1.mint(address(exchange), 10000e18);

        console.log("Additional token0 from swap: 50 token0");

        exchange.setManipulatedPrice(true);

        console.log("- Leverager paid %d%% more than fair price", ((manipulatedSpend * 100) / token1Spent) - 100);


        assertEq(token1Spent, 50e18, "Normal swap should cost 50 token1");
        assertEq(manipulatedSpend, 150e18, "Manipulated swap should cost 150 token1");
        assertEq(manipulatedSpend / token1Spent, 3, "Manipulation caused 3x price increase");
    }
```
